    52→    52→function useSymbols(ex: ExchangeId, market: MarketType | 'ALL') {
    53→    53→  const [rows, setRows] = useState<SymbolRow[]>([])
    54→    54→  const [loading, setLoading] = useState(true)
    55→    55→  useEffect(() => {
    56→    56→    let alive = true
    57→    57→    const hub = getHub()
    58→    58→    const load = async () => {
    59→    59→      // первый await выводит из синхронного контекста эффекта
    60→    60→      await Promise.resolve()
    61→    61→      if (!alive) return
    62→    62→      setLoading(true)
    63→    63→      if (market === 'ALL') {
    64→    64→        const [f, s] = await Promise.all([hub.fetchSymbols(ex, 'F'), hub.fetchSymbols(ex, 'S')])
    65→    65→        if (!alive) return
    66→    66→        const map = new Map<string, SymbolRow>()
    67→    67→        for (const r of [...f, ...s]) {
    68→    68→          const prev = map.get(r.symbol)
    69→    69→          if (!prev || r.vol24h > prev.vol24h) map.set(r.symbol, r)
    70→    70→        }
    71→    71→        setRows([...map.values()].sort((a, b) => b.vol24h - a.vol24h))
    72→    72→      } else {
    73→    73→        setRows(await hub.fetchSymbols(ex, market))
    74→    74→      }
    75→    75→      if (alive) setLoading(false)
    76→    76→    }
    77→    77→    void load()
    78→    78→    return () => { alive = false }
    79→    79→  }, [ex, market])
    80→    80→  return { rows, loading }
    81→    81→}

   745→   745→      </Dialog>

   750→   750→function DensityRow({ e, compact, onSend, tiers }: { e: DensityEvent; compact: boolean; onSend: () => void; tiers: VolumeTier[] | undefined }) {
