    27→function clamp(v: number, a: number, b: number): number {
    28→  return Math.max(a, Math.min(b, v))

    31→function niceStep(range: number, n: number): number {
    32→  const raw = range / Math.max(1, n)
    33→  if (!Number.isFinite(raw) || raw <= 0) return 1
    34→  const mag = Math.pow(10, Math.floor(Math.log10(raw)))
    35→  const norm = raw / mag
    36→  const step = norm < 1.5 ? 1 : norm < 3 ? 2 : norm < 7 ? 5 : 10
    37→  return step * mag
    38→}

    40→export function fmtChartPrice(p: number): string {
    41→  const a = Math.abs(p)
    42→  if (a >= 10000) return p.toLocaleString('en-US', { maximumFractionDigits: 1 })
    43→  if (a >= 100) return p.toFixed(2)
    44→  if (a >= 1) return p.toFixed(3)
    45→  if (a >= 0.01) return p.toFixed(5)
    46→  if (a >= 0.0001) return p.toFixed(7)
    47→  return p.toExponential(3)
    48→}

    50→function fmtAxisTime(t: number, interval: string): string {
    51→  const d = new Date(t)
    52→  const hhmm = `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    53→  if (interval === '1d' || interval === '1w') {
    54→    return d.toLocaleDateString('ru-RU', { day: '2-digit', month: 'short' })
    55→  }
    56→  if (interval === '1h' || interval === '4h') {
    57→    return `${d.toLocaleDateString('ru-RU', { day: '2-digit', month: 'short' })} ${hhmm}`
    58→  }
    59→  return hhmm
    60→}

    62→function fmtDuration(ms: number): string {
    63→  const s = Math.round(ms / 1000)
    64→  if (s < 90) return `${s}с`
    65→  const m = Math.round(s / 60)
    66→  if (m < 90) return `${m}м`
    67→  const h = Math.round(m / 60)
    68→  if (h < 36) return `${h}ч`
    69→  return `${Math.round(h / 24)}д`
    70→}

   165→export function CandleChart({
   166→  candles, interval, chartKey, tool = 'cursor', onToolDone, livePrice, chg24,
   167→  theme, className, showVolume = true, interactive = true, clearTick,
   168→  label, padLeft = 6,
   169→}: CandleChartProps) {
   170→  const wrapRef = useRef<HTMLDivElement>(null)
   171→  const canvasRef = useRef<HTMLCanvasElement>(null)
   172→  const viewRef = useRef<ViewBox>({ bars: 130, offset: -6, pRange: null })
   173→  const magnetRef = useRef(false)
   174→  const mouseRef = useRef<MouseState>({ x: 0, y: 0, inside: false, mode: 'idle', startX: 0, startY: 0, startOffset: 0 })
   175→  const drawingsRef = useRef<ChartDrawing[]>([])
   176→  const rulerRef = useRef<{ t1: number; p1: number; t2: number; p2: number } | null>(null)
   177→  const candlesRef = useRef<Candle[]>(candles)
   178→  const toolRef = useRef(tool)
   179→  const sizeRef = useRef({ w: 0, h: 0 })
   180→
   181→  // зеркала пропсов в рефах — только внутри эффектов
   182→  useEffect(() => { candlesRef.current = candles }, [candles])
   183→  useEffect(() => { toolRef.current = tool }, [tool])
   184→
   185→  // ── Цвета из CSS-переменных темы ──────────────────────────────────
   186→  const readColors = useCallback(() => {
   187→    if (typeof window === 'undefined') {
   188→      return { bg: '#0b0e17', fg: '#e2e8f0', grid: 'rgba(255,255,255,0.06)', muted: '#7c8aa5', border: 'rgba(255,255,255,0.1)', pos: '#34d399', neg: '#f43f5e', acc: '#8b5cf6', card: '#111524' }
   189→    }
   190→    const cs = getComputedStyle(document.documentElement)
   191→    const v = (n: string, f: string) => cs.getPropertyValue(n).trim() || f
   192→    return {
   193→      bg: v('--card-solid', '#0b0e17'),
   194→      fg: v('--foreground', '#e2e8f0'),
   195→      grid: v('--border', 'rgba(255,255,255,0.1)'),
   196→      muted: v('--muted-foreground', '#7c8aa5'),
   197→      border: v('--border', 'rgba(255,255,255,0.1)'),
   198→      pos: v('--pos', '#34d399'),
   199→      neg: v('--neg', '#f43f5e'),
   200→      acc: v('--acc', '#8b5cf6'),
   201→      card: v('--card-solid', '#111524'),
   202→    }
   203→  }, [])
   204→
   205→  // ── Геометрия ─────────────────────────────────────────────────────
   206→  const geom = useCallback(() => {
   207→    const w = sizeRef.current.w
   208→    const h = sizeRef.current.h
   209→    const padL = padLeft, padR = 64, padT = 10, padB = 20
   210→    const plotW = Math.max(30, w - padL - padR)
   211→    const fullH = Math.max(40, h - padT - padB)
   212→    const volH = showVolume ? fullH * 0.16 : 0
   213→    const chartH = fullH - volH - (showVolume ? 8 : 0)
   214→    return { padL, padR, padT, padB, plotW, fullH, volH, chartH }
   215→  }, [padLeft, showVolume])
   216→
   217→  // ── Единая шкала: видимые бары + диапазон цен ──────────────────
   218→  // Единственная точка правды для отрисовки и всех обработчиков:
   219→  // горизонтальный сдвиг (offset), ручная вертикальная панорама (pRange,
   220→  // включается после вертикального драга) и живая цена.
   221→  const computeScale = useCallback((): Scale | null => {
   222→    const cs = candlesRef.current
   223→    if (!cs || cs.length < 2) return null
   224→    const view = viewRef.current
   225→    const g = geom()
   226→    const intervalMs = cs[1].t - cs[0].t || 60_000
   227→    const len = cs.length
   228→    view.bars = clamp(view.bars, 15, Math.max(60, len + 40))
   229→    const rightIdx = clamp(len - view.offset, Math.max(5, view.bars * 0.35), len + view.bars * 0.35)
   230→    const startIdx = rightIdx - view.bars
   231→    const i0 = Math.max(0, Math.floor(startIdx))
   232→    const i1 = Math.min(len - 1, Math.ceil(rightIdx))
   233→    let minP = Infinity, maxP = -Infinity, maxV = 0
   234→    for (let i = i0; i <= i1; i++) {
   235→      const cd = cs[i]
   236→      if (cd.l < minP) minP = cd.l
   237→      if (cd.h > maxP) maxP = cd.h
   238→      if (cd.v > maxV) maxV = cd.v
   239→    }
   240→    if (!Number.isFinite(minP) || !Number.isFinite(maxP)) {
   241→      minP = 0
   242→      maxP = 1
   243→    }
   244→    if (livePrice && livePrice > 0) {
   245→      if (livePrice < minP) minP = livePrice
   246→      if (livePrice > maxP) maxP = livePrice
   247→    }
   248→    const padP = (maxP - minP) * 0.07 || Math.abs(maxP) * 0.01 || 1
   249→    let lo = minP - padP
   250→    let hi = maxP + padP
   251→    if (view.pRange) {
   252→      lo = view.pRange.min
   253→      hi = view.pRange.max
   254→    }
   255→    if (!(hi > lo)) hi = lo + Math.abs(lo) * 0.02 + 1
   256→    const barW = g.plotW / view.bars
   257→    const xOfIdx = (i: number) => g.padL + (i - startIdx + 0.5) * barW
   258→    const idxOfT = (t: number) => (t - cs[0].t) / intervalMs
   259→    return {
   260→      cs, intervalMs, len, rightIdx, startIdx, i0, i1, minP: lo, maxP: hi, maxV, barW,
   261→      padL: g.padL, padR: g.padR, padT: g.padT, padB: g.padB,
   262→      plotW: g.plotW, fullH: g.fullH, volH: g.volH, chartH: g.chartH,
   263→      xOfIdx, idxOfT,
   264→      xOfT: (t: number) => xOfIdx(idxOfT(t)),
   265→      yOfP: (p: number) => g.padT + ((hi - p) / (hi - lo)) * g.chartH,
   266→      pOfY: (y: number) => hi - ((y - g.padT) / g.chartH) * (hi - lo),
   267→      tOfX: (x: number) => cs[0].t + ((x - g.padL) / barW + startIdx - 0.5) * intervalMs,
   268→    }
   269→  }, [geom, livePrice])
   270→
   271→  // магнит (Ctrl): привязка точки к OHLC ближайшей свечи
   272→  const snapPoint = useCallback((t: number, p: number): { t: number; p: number } => {
   273→    const cs = candlesRef.current
   274→    if (!magnetRef.current || !cs || cs.length < 2) return { t, p }
   275→    const intervalMs = cs[1].t - cs[0].t || 60_000
   276→    const idx = clamp(Math.round((t - cs[0].t) / intervalMs), 0, cs.length - 1)
   277→    const cd = cs[idx]
   278→    let best = cd.c
   279→    for (const v of [cd.o, cd.h, cd.l, cd.c]) {
   280→      if (Math.abs(v - p) < Math.abs(best - p)) best = v
   281→    }
   282→    return { t: cd.t, p: best }
   283→  }, [])
   284→
   285→  // ── Главная отрисовка ─────────────────────────────────────────────
   286→  const redraw = useCallback(() => {
   287→    const cv = canvasRef.current
   288→    if (!cv) return
   289→    const ctx = cv.getContext('2d')
   290→    if (!ctx) return
   291→    const dpr = window.devicePixelRatio || 1
   292→    const w = sizeRef.current.w
   293→    const h = sizeRef.current.h
   294→    if (cv.width !== Math.round(w * dpr) || cv.height !== Math.round(h * dpr)) {
   295→      cv.width = Math.round(w * dpr)
   296→      cv.height = Math.round(h * dpr)
   297→    }
   298→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   299→    const C = readColors()
   300→    ctx.clearRect(0, 0, w, h)
   301→    ctx.fillStyle = C.bg
   302→    ctx.fillRect(0, 0, w, h)
   303→
   304→    const sc = computeScale()
   305→    if (!sc) {
   306→      ctx.fillStyle = C.muted
   307→      ctx.font = '12px ui-monospace, Menlo, monospace'
   308→      ctx.textAlign = 'center'
   309→      ctx.fillText('Загрузка свечей…', w / 2, h / 2)
   310→      return
   311→    }
   312→    const cs = sc.cs
   313→    const {
   314→      padL, padR, padT, plotW, fullH, volH, chartH, barW,
   315→      i0, i1, minP, maxP, maxV, len, intervalMs, startIdx,
   316→    } = sc
   317→    const xOfIdx = sc.xOfIdx
   318→    const xOfT = sc.xOfT
   319→    const yOfP = sc.yOfP
   320→    const pOfY = sc.pOfY
   321→    const tOfX = sc.tOfX
   322→
   323→    // сетка + шкала цены
   324→    ctx.font = '10px ui-monospace, Menlo, monospace'
   325→    ctx.textBaseline = 'middle'
   326→    const step = niceStep(maxP - minP, 6)
   327→    ctx.strokeStyle = C.grid
   328→    ctx.lineWidth = 1
   329→    ctx.fillStyle = C.muted
   330→    ctx.textAlign = 'left'
   331→    for (let p = Math.ceil(minP / step) * step; p <= maxP; p += step) {
   332→      const y = yOfP(p)
   333→      ctx.beginPath()
   334→      ctx.moveTo(padL, Math.round(y) + 0.5)
   335→      ctx.lineTo(padL + plotW, Math.round(y) + 0.5)
   336→      ctx.stroke()
   337→      const ap = fmtChartPrice(p)
   338→      if (ctx.measureText(ap).width <= padR - 4) ctx.fillText(ap, padL + plotW + 6, y)
   339→    }
   340→
   341→    // шкала времени (смена дня/месяца — яркая метка с датой)
   342→    const labelEvery = Math.max(1, Math.ceil(92 / barW))
   343→    ctx.textAlign = 'center'
   344→    for (let i = i0; i <= i1; i++) {
   345→      if (i % labelEvery !== 0) continue
   346→      const x = xOfIdx(i)
   347→      const d = new Date(cs[i].t)
   348→      const prev = i > 0 ? new Date(cs[i - 1].t) : null
   349→      const longTf = interval === '1d' || interval === '1w'
   350→      const boundary = prev
   351→        ? (longTf ? prev.getMonth() !== d.getMonth() : prev.getDate() !== d.getDate())
   352→        : false
   353→      ctx.strokeStyle = boundary ? `${C.muted}44` : C.grid
   354→      ctx.lineWidth = 1
   355→      ctx.beginPath()
   356→      ctx.moveTo(Math.round(x) + 0.5, padT)
   357→      ctx.lineTo(Math.round(x) + 0.5, padT + fullH)
   358→      ctx.stroke()
   359→      ctx.fillStyle = boundary ? C.fg : C.muted
   360→      ctx.font = boundary
   361→        ? 'bold 10px ui-monospace, Menlo, monospace'
   362→        : '10px ui-monospace, Menlo, monospace'
   363→      ctx.fillText(
   364→        boundary
   365→          ? d.toLocaleDateString('ru-RU', longTf ? { month: 'short', year: '2-digit' } : { day: '2-digit', month: 'short' })
   366→          : fmtAxisTime(cs[i].t, interval),
   367→        x, padT + fullH + 11,
   368→      )
   369→    }
   370→
   371→    // водяной знак тикера по центру
   372→    if (label && plotW > 300) {
   373→      const wm = label.split(' · ')[2] ?? label
   374→      ctx.save()
   375→      ctx.globalAlpha = 0.055
   376→      ctx.font = `bold ${Math.round(clamp(plotW / 9, 28, 64))}px ui-monospace, Menlo, monospace`
   377→      ctx.fillStyle = C.fg
   378→      ctx.textAlign = 'center'
   379→      ctx.fillText(wm, padL + plotW / 2, padT + chartH / 2 + 12)
   380→      ctx.restore()
   381→    }
   382→
   383→    // объёмы — с вертикальным градиентом
   384→    if (showVolume && maxV > 0) {
   385→      const volTop = padT + chartH + 8
   386→      const gUp = ctx.createLinearGradient(0, volTop, 0, volTop + volH)
   387→      gUp.addColorStop(0, `${C.pos}59`)
   388→      gUp.addColorStop(1, `${C.pos}12`)
   389→      const gDn = ctx.createLinearGradient(0, volTop, 0, volTop + volH)
   390→      gDn.addColorStop(0, `${C.neg}59`)
   391→      gDn.addColorStop(1, `${C.neg}12`)
   392→      for (let i = i0; i <= i1; i++) {
   393→        const cd = cs[i]
   394→        const x = xOfIdx(i)
   395→        const vh = (cd.v / maxV) * volH
   396→        ctx.fillStyle = cd.c >= cd.o ? gUp : gDn
   397→        ctx.fillRect(Math.round(x - barW * 0.33), volTop + volH - vh, Math.max(1, Math.round(barW * 0.66)), vh)
   398→      }
   399→    }
   400→
   401→    // свечи — хвост в пиксельной сетке, тело с капом ширины
   402→    const bodyW = clamp(barW * 0.66, 1, 21)
   403→    for (let i = i0; i <= i1; i++) {
   404→      const cd = cs[i]
   405→      const x = xOfIdx(i)
   406→      const up = cd.c >= cd.o
   407→      const col = up ? C.pos : C.neg
   408→      const wx = Math.round(x) + 0.5
   409→      ctx.strokeStyle = col
   410→      ctx.lineWidth = 1
   411→      ctx.beginPath()
   412→      ctx.moveTo(wx, yOfP(cd.h))
   413→      ctx.lineTo(wx, yOfP(cd.l))
   414→      ctx.stroke()
   415→      const yO = yOfP(cd.o)
   416→      const yC = yOfP(cd.c)
   417→      const top = Math.min(yO, yC)
   418→      const bh = Math.max(1, Math.abs(yC - yO))
   419→      ctx.fillStyle = col
   420→      ctx.fillRect(Math.round(x - bodyW / 2), Math.round(top), Math.round(bodyW), Math.max(1, Math.round(bh)))
   421→    }
   422→
   423→    // метки hi/lo видимого диапазона
   424→    if (i1 > i0 + 2) {
   425→      let hiI = i0
   426→      let loI = i0
   427→      for (let i = i0 + 1; i <= i1; i++) {
   428→        if (cs[i].h > cs[hiI].h) hiI = i
   429→        if (cs[i].l < cs[loI].l) loI = i
   430→      }
   431→      const mark = (idx: number, p: number, above: boolean) => {
   432→        const x = xOfIdx(idx)
   433→        const y = yOfP(p)
   434→        ctx.font = 'bold 9px ui-monospace, Menlo, monospace'
   435→        const txt = fmtChartPrice(p)
   436→        const w = ctx.measureText(txt).width + 10
   437→        const yy = above ? y - 19 : y + 5
   438→        roundRect(ctx, clamp(x - w / 2, padL, padL + plotW - w), yy, w, 14, 3)
   439→        ctx.fillStyle = `${C.muted}e6`
   440→        ctx.fill()
   441→        ctx.fillStyle = C.fg
   442→        ctx.textAlign = 'center'
   443→        ctx.fillText(txt, clamp(x, padL + w / 2, padL + plotW - w / 2), yy + 7)
   444→      }
   445→      mark(hiI, cs[hiI].h, true)
   446→      mark(loI, cs[loI].l, false)
   447→    }

   792→    const near = (ax: number, ay: number, bx: number, by: number): boolean => {
   793→      const dx = bx - ax, dy = by - ay
   794→      const l2 = dx * dx + dy * dy
   795→      if (l2 === 0) return Math.hypot(x - ax, y - ay) < 7
   796→      let t = ((x - ax) * dx + (y - ay) * dy) / l2
   797→      t = clamp(t, 0, 1)

   804→        if (near(pts[0].x, pts[0].y, pts[1].x, pts[1].y)) return d.id
   805→      } else if (d.type === 'hline' && pts.length >= 1) {

   807→      } else if (d.type === 'vline' && pts.length >= 1) {

   809→      } else if (d.type === 'rect' && pts.length >= 2) {
   810→        const x1 = Math.min(pts[0].x, pts[1].x), x2 = Math.max(pts[0].x, pts[1].x)
   811→        const y1 = Math.min(pts[0].y, pts[1].y), y2 = Math.max(pts[0].y, pts[1].y)
   812→        if (near(x1, y1, x2, y1) || near(x1, y2, x2, y2) || near(x1, y1, x1, y2) || near(x2, y1, x2, y2)) return d.id
   813→      } else if (d.type === 'fib' && pts.length >= 2) {

   858→    })
   859→    ro.observe(wrap)
   860→    const r = wrap.getBoundingClientRect()
   861→    sizeRef.current = { w: r.width, h: r.height }
   862→    redrawRef.current()
   863→    return () => ro.disconnect()
   864→  }, [])
   865→
   866→  // Esc — отмена рисования / линейка / курсор
   867→  useEffect(() => {
   868→    const onKey = (e: KeyboardEvent) => {
   869→      if (e.key !== 'Escape') return
   870→      const ms = mouseRef.current
   871→      if (ms.mode === 'draw2') {
   872→        ms.mode = 'idle'
   873→        ms.pending = undefined
   874→        onToolDone?.()
   875→        redrawRef.current()
   876→      } else if (rulerRef.current || ms.mode === 'ruler') {
   877→        rulerRef.current = null
   878→        ms.mode = 'idle'
   879→        ms.pending = undefined
   880→        redrawRef.current()
   881→      } else if (toolRef.current !== 'cursor') {
   882→        onToolDone?.()
   883→      }
   884→    }
   885→    window.addEventListener('keydown', onKey)
   886→    return () => window.removeEventListener('keydown', onKey)
   887→  }, [onToolDone])
   888→
   889→  // Ctrl — магнит: привязка рисовалок/линейки к OHLC свечей, пока зажат
   890→  useEffect(() => {
   891→    if (!interactive) return
   892→    const apply = (down: boolean, e: KeyboardEvent) => {
   893→      if (e.key !== 'Control' || magnetRef.current === down) return
   894→      magnetRef.current = down
   895→      redrawRef.current()
   896→    }
   897→    const onKeyDown = (e: KeyboardEvent) => apply(true, e)
   898→    const onKeyUp = (e: KeyboardEvent) => apply(false, e)
   899→    const onBlur = () => {
   900→      if (!magnetRef.current) return
   901→      magnetRef.current = false
   902→      redrawRef.current()
   903→    }
   904→    window.addEventListener('keydown', onKeyDown)
   905→    window.addEventListener('keyup', onKeyUp)
   906→    window.addEventListener('blur', onBlur)
   907→    return () => {
   908→      window.removeEventListener('keydown', onKeyDown)
   909→      window.removeEventListener('keyup', onKeyUp)
   910→      window.removeEventListener('blur', onBlur)
   911→    }
   912→  }, [interactive])
   913→
   914→  // отпустили колёсико мыши (в любой точке окна) — линейка выключается
   915→  useEffect(() => {
   916→    if (!interactive) return
   917→    const onWinMouseUp = () => {
   918→      const ms = mouseRef.current
   919→      if (ms.mode !== 'ruler') return
   920→      ms.mode = 'idle'
   921→      ms.pending = undefined
   922→      rulerRef.current = null
   923→      redrawRef.current()
   924→    }
   925→    window.addEventListener('mouseup', onWinMouseUp)
   926→    return () => window.removeEventListener('mouseup', onWinMouseUp)
   927→  }, [interactive])
   928→
   929→  // ── Обработчики мыши ──────────────────────────────────────────────
   930→  const pos = (e: React.MouseEvent | MouseEvent) => {
   931→    const cv = canvasRef.current!
   932→    const r = cv.getBoundingClientRect()
   933→    return { x: e.clientX - r.left, y: e.clientY - r.top }
   934→  }
   935→
   936→  const onMouseMove = (e: React.MouseEvent) => {
   937→    const ms = mouseRef.current
   938→    const p = pos(e)
   939→    ms.x = p.x
   940→    ms.y = p.y
   941→    ms.inside = true
   942→    const cv = canvasRef.current
   943→    if (!cv) return
   944→
   945→    if (ms.mode === 'pan') {
   946→      const sc = computeScale()
   947→      if (sc) {
   948→        const view = viewRef.current
   949→        view.offset = ms.startOffset + (p.x - ms.startX) / sc.barW
   950→        // вертикальная панорама: диапазон цен следует за курсором
   951→        if (ms.startPRange && Math.abs(p.y - ms.startY) > 3) {
   952→          const range = ms.startPRange.max - ms.startPRange.min
   953→          if (range > 0 && Number.isFinite(range)) {
   954→            const shift = ((p.y - ms.startY) / sc.chartH) * range
   955→            view.pRange = { min: ms.startPRange.min + shift, max: ms.startPRange.max + shift }
   956→          }
   957→        }
   958→      }
   959→    } else if (ms.mode === 'drag' && ms.grab) {
   960→      const sc = computeScale()
   961→      if (sc) {
   962→        const dIdx = (p.x - ms.startX) / sc.barW
   963→        const dPrice = sc.pOfY(p.y) - ms.grab.priceStart
   964→        const dTime = dIdx * sc.intervalMs
   965→        const target = drawingsRef.current.find((d) => d.id === ms.grab!.id)
   966→        if (target) {
   967→          target.points = ms.grab.orig.map((pt) => ({ t: pt.t + dTime, p: pt.p + dPrice }))
   968→          if (chartKey) saveDrawings(chartKey, drawingsRef.current)
   969→        }
   970→      }
   971→    } else if (ms.mode === 'idle') {
   972→      ms.hoverId = hitTest(p.x, p.y) ?? undefined
   973→      cv.style.cursor = ms.hoverId ? 'pointer' : 'crosshair'
   974→    }
   975→    redraw()
   976→  }
   977→
   978→  const onMouseDown = (e: React.MouseEvent) => {
   979→    if (!interactive) return
   980→    const ms = mouseRef.current
   981→    const p = pos(e)
   982→    const cv = canvasRef.current
   983→    if (!cv) return
   984→    const sc = computeScale()
   985→    if (!sc) return
   986→    const anchor = snapPoint(sc.tOfX(p.x), sc.pOfY(p.y))
   987→
   988→    // средняя кнопка — линейка (живёт, пока колёсико зажато)
   989→    if (e.button === 1) {
   990→      e.preventDefault()
   991→      ms.mode = 'ruler'
   992→      ms.pending = anchor
   993→      redraw()
   994→      return
   995→    }
   996→    // правая кнопка — удалить рисунок под курсором
   997→    if (e.button === 2) {
   998→      const id = hitTest(p.x, p.y)
   999→      if (id) {
  1000→        drawingsRef.current = drawingsRef.current.filter((d) => d.id !== id)
  1001→        if (chartKey) saveDrawings(chartKey, drawingsRef.current)
  1002→        redraw()
  1003→      }
  1004→      return
  1005→    }
  1006→    if (e.button !== 0) return
  1007→
  1008→    const t = toolRef.current
  1009→    if (t === 'cursor') {
  1010→      const id = hitTest(p.x, p.y)
  1011→      if (id && chartKey) {
  1012→        const target = drawingsRef.current.find((d) => d.id === id)
  1013→        if (target) {
  1014→          ms.mode = 'drag'
  1015→          ms.grab = { id, orig: target.points.map((pt) => ({ ...pt })), idxStart: sc.idxOfT(anchor.t), priceStart: anchor.p }
  1016→          ms.startX = p.x
  1017→          ms.startY = p.y
  1018→        }
  1019→      } else {
  1020→        ms.mode = 'pan'
  1021→        ms.startX = p.x
  1022→        ms.startY = p.y
  1023→        ms.startOffset = viewRef.current.offset
  1024→        ms.startPRange = { min: sc.minP, max: sc.maxP }
  1025→        cv.style.cursor = 'grabbing'
  1026→      }
  1027→      redraw()
  1028→      return
  1029→    }
  1030→    if (t === 'ruler') {
  1031→      if (ms.mode === 'draw2' && ms.pending) {
  1032→        rulerRef.current = { t1: ms.pending.t, p1: ms.pending.p, t2: anchor.t, p2: anchor.p }
  1033→        ms.mode = 'idle'
  1034→        ms.pending = undefined
  1035→        onToolDone?.()
  1036→      } else {
  1037→        ms.mode = 'draw2'
  1038→        ms.pending = anchor
  1039→      }
  1040→      redraw()
  1041→      return
  1042→    }
  1043→    if (t === 'hline' || t === 'vline') {
  1044→      if (chartKey) {
  1045→        drawId += 1
  1046→        drawingsRef.current = [...drawingsRef.current, { id: `d${Date.now()}_${drawId}`, type: t, points: [anchor] }]
  1047→        saveDrawings(chartKey, drawingsRef.current)
  1048→      }
  1049→      onToolDone?.()
  1050→      redraw()
  1051→      return
  1052→    }
  1053→    // двухточечные: тренд / прямоугольник / фибо
  1054→    if (ms.mode === 'draw2' && ms.pending) {
  1055→      if (chartKey) {
  1056→        drawId += 1
  1057→        drawingsRef.current = [...drawingsRef.current, { id: `d${Date.now()}_${drawId}`, type: t as DrawType, points: [ms.pending, anchor] }]
  1058→        saveDrawings(chartKey, drawingsRef.current)
  1059→      }
  1060→      ms.mode = 'idle'
  1061→      ms.pending = undefined
  1062→      onToolDone?.()
  1063→    } else {
  1064→      ms.mode = 'draw2'
  1065→      ms.pending = anchor
  1066→    }
  1067→    redraw()
  1068→  }
  1069→
  1070→  const onMouseUp = () => {
  1071→    const ms = mouseRef.current
  1072→    // линейка колёсиком выключается на window.mouseup — отпустил, она исчезла
  1073→    if (ms.mode === 'pan' || ms.mode === 'drag') {
  1074→      ms.mode = 'idle'
  1075→      ms.grab = undefined
  1076→      const cv = canvasRef.current
  1077→      if (cv) cv.style.cursor = 'crosshair'
  1078→    }
  1079→    redraw()
  1080→  }
  1081→
  1082→  const onWheelNative = useCallback((e: WheelEvent) => {
  1083→    if (!interactive) return
  1084→    e.preventDefault()
  1085→    const sc = computeScale()
  1086→    if (!sc) return
  1087→    const cv = canvasRef.current
  1088→    if (!cv) return
  1089→    const r = cv.getBoundingClientRect()
  1090→    const px = e.clientX - r.left
  1091→    const factor = e.deltaY > 0 ? 1.15 : 1 / 1.15
  1092→    const view = viewRef.current
  1093→    const newBars = clamp(view.bars * factor, 15, Math.max(60, sc.len + 40))
  1094→    // якорь: индекс под курсором остаётся на месте
  1095→    const anchorIdx = sc.startIdx + ((px - sc.padL) / sc.plotW) * view.bars
  1096→    const newRight = anchorIdx + (1 - (px - sc.padL) / sc.plotW) * newBars
  1097→    view.bars = newBars
  1098→    view.offset = sc.len - newRight
  1099→    redrawRef.current()
  1100→  }, [computeScale, interactive])
  1101→
  1102→  // нативный non-passive wheel: зум колесом без прокрутки страницы
  1103→  useEffect(() => {
  1104→    const cv = canvasRef.current
  1105→    if (!cv) return
  1106→    cv.addEventListener('wheel', onWheelNative, { passive: false })
  1107→    return () => cv.removeEventListener('wheel', onWheelNative)
  1108→  }, [onWheelNative])
  1109→
  1110→  const onDoubleClick = () => {
  1111→    viewRef.current = { bars: 130, offset: -6, pRange: null }
  1112→    redraw()
  1113→  }
  1114→
  1115→  const onLeave = () => {
  1116→    const ms = mouseRef.current
  1117→    ms.inside = false
  1118→    if (ms.mode === 'pan' || ms.mode === 'drag') {
  1119→      ms.mode = 'idle'
  1120→      ms.grab = undefined
  1121→      const cv = canvasRef.current
  1122→      if (cv) cv.style.cursor = 'crosshair'
  1123→    }
  1124→    // линейку колёсика не отменяем на выходе из канваса —
  1125→    // она выключится, когда колёсико будет отпущено (window mouseup)
  1126→    redraw()
  1127→  }
  1128→
  1129→  return (
  1130→    <div ref={wrapRef} className={cn('relative h-full w-full', className)}>
  1131→      <canvas
  1132→        ref={canvasRef}
  1133→        className="absolute inset-0 h-full w-full"
  1134→        style={{ touchAction: 'none' }}
  1135→        onMouseMove={onMouseMove}
  1136→        onMouseDown={onMouseDown}
  1137→        onMouseUp={onMouseUp}
  1138→        onMouseLeave={onLeave}
  1139→        onDoubleClick={onDoubleClick}
  1140→        onContextMenu={(e) => e.preventDefault()}
  1141→      />
  1142→    </div>
  1143→  )
  1144→}
