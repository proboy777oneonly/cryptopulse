    27→function clamp(v: number, a: number, b: number): number {
    28→  return Math.max(a, Math.min(b, v))

    31→function niceStep(range: number, n: number): number {
    32→  const raw = range / Math.max(1, n)
    33→  if (!Number.isFinite(raw) || raw <= 0) return 1
    34→  const mag = Math.pow(10, Math.floor(Math.log10(raw)))
    35→  const norm = raw / mag
    36→  const step = norm < 1.5 ? 1 : norm < 3 ? 2 : norm < 7 ? 5 : 10
    37→  return step * mag
    38→}

    40→export function fmtChartPrice(p: number): string {
    41→  const a = Math.abs(p)
    42→  if (a >= 10000) return p.toLocaleString('en-US', { maximumFractionDigits: 1 })
    43→  if (a >= 100) return p.toFixed(2)
    44→  if (a >= 1) return p.toFixed(3)
    45→  if (a >= 0.01) return p.toFixed(5)
    46→  if (a >= 0.0001) return p.toFixed(7)
    47→  return p.toExponential(3)
    48→}

    50→function fmtAxisTime(t: number, interval: string): string {
    51→  const d = new Date(t)
    52→  const hhmm = `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    53→  if (interval === '1d' || interval === '1w') {
    54→    return d.toLocaleDateString('ru-RU', { day: '2-digit', month: 'short' })
    55→  }
    56→  if (interval === '1h' || interval === '4h') {
    57→    return `${d.toLocaleDateString('ru-RU', { day: '2-digit', month: 'short' })} ${hhmm}`
    58→  }
    59→  return hhmm
    60→}

    62→function fmtDuration(ms: number): string {
    63→  const s = Math.round(ms / 1000)
    64→  if (s < 90) return `${s}с`
    65→  const m = Math.round(s / 60)
    66→  if (m < 90) return `${m}м`
    67→  const h = Math.round(m / 60)
    68→  if (h < 36) return `${h}ч`
    69→  return `${Math.round(h / 24)}д`
    70→}

   124→function loadDrawings(key: string): ChartDrawing[] {
   125→  try {
   126→    const raw = localStorage.getItem(`cp-draw-${key}`)
   127→    if (!raw) return []
   128→    const arr = JSON.parse(raw) as ChartDrawing[]
   129→    return Array.isArray(arr) ? arr.filter((d) => d && d.type && Array.isArray(d.points)) : []

   131→    return []
   132→  }

   163→export function CandleChart({
   164→  candles, interval, chartKey, tool = 'cursor', onToolDone, livePrice,
   165→  theme, className, showVolume = true, interactive = true, clearTick,
   166→  label, padLeft = 6,
   167→}: CandleChartProps) {
   168→  const wrapRef = useRef<HTMLDivElement>(null)
   169→  const canvasRef = useRef<HTMLCanvasElement>(null)
   170→  const viewRef = useRef<ViewBox>({ bars: 130, offset: 0, pRange: null })
   171→  const magnetRef = useRef(false)
   172→  const mouseRef = useRef<MouseState>({ x: 0, y: 0, inside: false, mode: 'idle', startX: 0, startY: 0, startOffset: 0 })
   173→  const drawingsRef = useRef<ChartDrawing[]>([])
   174→  const rulerRef = useRef<{ t1: number; p1: number; t2: number; p2: number } | null>(null)
   175→  const candlesRef = useRef<Candle[]>(candles)
   176→  const toolRef = useRef(tool)
   177→  const sizeRef = useRef({ w: 0, h: 0 })
   178→
   179→  // зеркала пропсов в рефах — только внутри эффектов
   180→  useEffect(() => { candlesRef.current = candles }, [candles])
   181→  useEffect(() => { toolRef.current = tool }, [tool])
   182→
   183→  // ── Цвета из CSS-переменных темы ──────────────────────────────────
   184→  const readColors = useCallback(() => {
   185→    if (typeof window === 'undefined') {
   186→      return { bg: '#0b0e17', fg: '#e2e8f0', grid: 'rgba(255,255,255,0.06)', muted: '#7c8aa5', border: 'rgba(255,255,255,0.1)', pos: '#34d399', neg: '#f43f5e', acc: '#8b5cf6', card: '#111524' }
   187→    }
   188→    const cs = getComputedStyle(document.documentElement)
   189→    const v = (n: string, f: string) => cs.getPropertyValue(n).trim() || f
   190→    return {
   191→      bg: v('--card-solid', '#0b0e17'),
   192→      fg: v('--foreground', '#e2e8f0'),
   193→      grid: v('--border', 'rgba(255,255,255,0.1)'),
   194→      muted: v('--muted-foreground', '#7c8aa5'),
   195→      border: v('--border', 'rgba(255,255,255,0.1)'),
   196→      pos: v('--pos', '#34d399'),
   197→      neg: v('--neg', '#f43f5e'),
   198→      acc: v('--acc', '#8b5cf6'),
   199→      card: v('--card-solid', '#111524'),
   200→    }
   201→  }, [])
   202→
   203→  // ── Геометрия ─────────────────────────────────────────────────────
   204→  const geom = useCallback(() => {
   205→    const w = sizeRef.current.w
   206→    const h = sizeRef.current.h
   207→    const padL = padLeft, padR = 64, padT = 10, padB = 20
   208→    const plotW = Math.max(30, w - padL - padR)
   209→    const fullH = Math.max(40, h - padT - padB)
   210→    const volH = showVolume ? fullH * 0.16 : 0
   211→    const chartH = fullH - volH - (showVolume ? 8 : 0)
   212→    return { padL, padR, padT, padB, plotW, fullH, volH, chartH }
   213→  }, [padLeft, showVolume])
   214→
   215→  // ── Единая шкала: видимые бары + диапазон цен ──────────────────
   216→  // Единственная точка правды для отрисовки и всех обработчиков:
   217→  // горизонтальный сдвиг (offset), ручная вертикальная панорама (pRange,
   218→  // включается после вертикального драга) и живая цена.
   219→  const computeScale = useCallback((): Scale | null => {
   220→    const cs = candlesRef.current
   221→    if (!cs || cs.length < 2) return null
   222→    const view = viewRef.current
   223→    const g = geom()
   224→    const intervalMs = cs[1].t - cs[0].t || 60_000
   225→    const len = cs.length
   226→    view.bars = clamp(view.bars, 15, Math.max(60, len + 40))
   227→    const rightIdx = clamp(len - view.offset, Math.max(5, view.bars * 0.35), len + view.bars * 0.35)
   228→    const startIdx = rightIdx - view.bars
   229→    const i0 = Math.max(0, Math.floor(startIdx))
   230→    const i1 = Math.min(len - 1, Math.ceil(rightIdx))
   231→    let minP = Infinity, maxP = -Infinity, maxV = 0
   232→    for (let i = i0; i <= i1; i++) {
   233→      const cd = cs[i]
   234→      if (cd.l < minP) minP = cd.l
   235→      if (cd.h > maxP) maxP = cd.h
   236→      if (cd.v > maxV) maxV = cd.v
   237→    }
   238→    if (!Number.isFinite(minP) || !Number.isFinite(maxP)) {
   239→      minP = 0
   240→      maxP = 1
   241→    }
   242→    if (livePrice && livePrice > 0) {
   243→      if (livePrice < minP) minP = livePrice
   244→      if (livePrice > maxP) maxP = livePrice
   245→    }
   246→    const padP = (maxP - minP) * 0.07 || Math.abs(maxP) * 0.01 || 1
   247→    let lo = minP - padP
   248→    let hi = maxP + padP
   249→    if (view.pRange) {
   250→      lo = view.pRange.min
   251→      hi = view.pRange.max
   252→    }
   253→    if (!(hi > lo)) hi = lo + Math.abs(lo) * 0.02 + 1
   254→    const barW = g.plotW / view.bars
   255→    const xOfIdx = (i: number) => g.padL + (i - startIdx + 0.5) * barW
   256→    const idxOfT = (t: number) => (t - cs[0].t) / intervalMs
   257→    return {
   258→      cs, intervalMs, len, rightIdx, startIdx, i0, i1, minP: lo, maxP: hi, maxV, barW,
   259→      padL: g.padL, padR: g.padR, padT: g.padT, padB: g.padB,
   260→      plotW: g.plotW, fullH: g.fullH, volH: g.volH, chartH: g.chartH,
   261→      xOfIdx, idxOfT,
   262→      xOfT: (t: number) => xOfIdx(idxOfT(t)),
   263→      yOfP: (p: number) => g.padT + ((hi - p) / (hi - lo)) * g.chartH,
   264→      pOfY: (y: number) => hi - ((y - g.padT) / g.chartH) * (hi - lo),
   265→      tOfX: (x: number) => cs[0].t + ((x - g.padL) / barW + startIdx - 0.5) * intervalMs,
   266→    }
   267→  }, [geom, livePrice])
   268→
   269→  // магнит (Ctrl): привязка точки к OHLC ближайшей свечи
   270→  const snapPoint = useCallback((t: number, p: number): { t: number; p: number } => {
   271→    const cs = candlesRef.current
   272→    if (!magnetRef.current || !cs || cs.length < 2) return { t, p }
   273→    const intervalMs = cs[1].t - cs[0].t || 60_000
   274→    const idx = clamp(Math.round((t - cs[0].t) / intervalMs), 0, cs.length - 1)
   275→    const cd = cs[idx]
   276→    let best = cd.c
   277→    for (const v of [cd.o, cd.h, cd.l, cd.c]) {
   278→      if (Math.abs(v - p) < Math.abs(best - p)) best = v
   279→    }
   280→    return { t: cd.t, p: best }
   281→  }, [])
   282→
   283→  // ── Главная отрисовка ─────────────────────────────────────────────
   284→  const redraw = useCallback(() => {
   285→    const cv = canvasRef.current
   286→    if (!cv) return
   287→    const ctx = cv.getContext('2d')
   288→    if (!ctx) return
   289→    const dpr = window.devicePixelRatio || 1
   290→    const w = sizeRef.current.w
   291→    const h = sizeRef.current.h
   292→    if (cv.width !== Math.round(w * dpr) || cv.height !== Math.round(h * dpr)) {
   293→      cv.width = Math.round(w * dpr)
   294→      cv.height = Math.round(h * dpr)
   295→    }
   296→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   297→    const C = readColors()
   298→    ctx.clearRect(0, 0, w, h)
   299→    ctx.fillStyle = C.bg
   300→    ctx.fillRect(0, 0, w, h)
   301→
   302→    const sc = computeScale()
   303→    if (!sc) {
   304→      ctx.fillStyle = C.muted
   305→      ctx.font = '12px ui-monospace, Menlo, monospace'
   306→      ctx.textAlign = 'center'
   307→      ctx.fillText('Загрузка свечей…', w / 2, h / 2)
   308→      return
   309→    }
   310→    const cs = sc.cs
   311→    const {
   312→      padL, padR, padT, plotW, fullH, volH, chartH, barW,
   313→      i0, i1, minP, maxP, maxV, len, intervalMs, startIdx,
   314→    } = sc
   315→    const xOfIdx = sc.xOfIdx
   316→    const xOfT = sc.xOfT
   317→    const yOfP = sc.yOfP
   318→    const pOfY = sc.pOfY
   319→    const tOfX = sc.tOfX
   320→
   321→    // сетка + шкала цены
   322→    ctx.font = '10px ui-monospace, Menlo, monospace'
   323→    ctx.textBaseline = 'middle'
   324→    const step = niceStep(maxP - minP, 6)
   325→    ctx.strokeStyle = C.grid
   326→    ctx.lineWidth = 1
   327→    ctx.fillStyle = C.muted
   328→    ctx.textAlign = 'left'
   329→    for (let p = Math.ceil(minP / step) * step; p <= maxP; p += step) {
   330→      const y = yOfP(p)
   331→      ctx.beginPath()
   332→      ctx.moveTo(padL, Math.round(y) + 0.5)
   333→      ctx.lineTo(padL + plotW, Math.round(y) + 0.5)
   334→      ctx.stroke()
   335→      ctx.fillText(fmtChartPrice(p), padL + plotW + 6, y)
   336→    }
   337→
   338→    // шкала времени
   339→    const labelEvery = Math.max(1, Math.ceil(92 / barW))
   340→    ctx.textAlign = 'center'
   341→    for (let i = i0; i <= i1; i++) {
   342→      if (i % labelEvery !== 0) continue
   343→      const x = xOfIdx(i)
   344→      ctx.strokeStyle = C.grid
   345→      ctx.beginPath()
   346→      ctx.moveTo(Math.round(x) + 0.5, padT)
   347→      ctx.lineTo(Math.round(x) + 0.5, padT + fullH)
   348→      ctx.stroke()
   349→      ctx.fillStyle = C.muted
   350→      ctx.fillText(fmtAxisTime(cs[i].t, interval), x, padT + fullH + 11)
   351→    }
   352→
   353→    // объёмы
   354→    if (showVolume && maxV > 0) {
   355→      const volTop = padT + chartH + 8
   356→      for (let i = i0; i <= i1; i++) {
   357→        const cd = cs[i]
   358→        const x = xOfIdx(i)
   359→        const vh = (cd.v / maxV) * volH
   360→        ctx.fillStyle = cd.c >= cd.o ? `${C.pos}33` : `${C.neg}33`
   361→        ctx.fillRect(x - barW * 0.33, volTop + volH - vh, Math.max(1, barW * 0.66), vh)
   362→      }
   363→    }
   364→
   365→    // свечи
   366→    const bodyW = Math.max(1, barW * 0.66)
   367→    for (let i = i0; i <= i1; i++) {
   368→      const cd = cs[i]
   369→      const x = xOfIdx(i)
   370→      const up = cd.c >= cd.o
   371→      const col = up ? C.pos : C.neg
   372→      ctx.strokeStyle = col
   373→      ctx.lineWidth = 1
   374→      ctx.beginPath()
   375→      ctx.moveTo(Math.round(x) + 0.5, yOfP(cd.h))
   376→      ctx.lineTo(Math.round(x) + 0.5, yOfP(cd.l))
   377→      ctx.stroke()
   378→      const yO = yOfP(cd.o)
   379→      const yC = yOfP(cd.c)
   380→      const top = Math.min(yO, yC)
   381→      const bh = Math.max(1, Math.abs(yC - yO))
   382→      ctx.fillStyle = col
   383→      ctx.fillRect(x - bodyW / 2, top, bodyW, bh)
   384→    }
   385→
   386→    // последняя цена (при ручной панораме чип прилипает к краю области)
   387→    const lastC = cs[len - 1]
   388→    const lastP = livePrice && livePrice > 0 ? livePrice : lastC.c
   389→    const lastUp = lastP >= lastC.o
   390→    const yLastRaw = yOfP(lastP)
   391→    const yLast = clamp(yLastRaw, padT + 8, padT + chartH - 8)
   392→    if (yLastRaw >= padT && yLastRaw <= padT + chartH) {
   393→      ctx.setLineDash([4, 4])
   394→      ctx.strokeStyle = lastUp ? C.pos : C.neg
   395→      ctx.lineWidth = 1
   396→      ctx.beginPath()
   397→      ctx.moveTo(padL, Math.round(yLast) + 0.5)
   398→      ctx.lineTo(padL + plotW, Math.round(yLast) + 0.5)
   399→      ctx.stroke()
   400→      ctx.setLineDash([])
   401→    }
   402→    // чип последней цены
   403→    ctx.fillStyle = lastUp ? C.pos : C.neg
   404→    const lpTxt = fmtChartPrice(lastP)
   405→    ctx.font = 'bold 10px ui-monospace, Menlo, monospace'
   406→    const lpW = ctx.measureText(lpTxt).width + 10
   407→    roundRect(ctx, padL + plotW + 2, yLast - 8, Math.min(lpW, padR - 4), 16, 4)
   408→    ctx.fill()
   409→    ctx.fillStyle = '#0b0e17'
   410→    ctx.textAlign = 'left'
   411→    ctx.fillText(lpTxt, padL + plotW + 7, yLast)
   412→
   413→    // легенда: подпись графика (биржа · рынок · тикер · ТФ) + OHLC
   414→    const ms = mouseRef.current
   415→    if (plotW >= 420) {
   416→      ctx.textAlign = 'left'
   417→      if (label) {
   418→        ctx.font = 'bold 10px ui-monospace, Menlo, monospace'
   419→        let txt = label
   420→        const maxLabelW = Math.max(140, plotW / 2 - 150)
   421→        while (ctx.measureText(txt).width > maxLabelW && txt.includes(' · ')) {
   422→          txt = txt.split(' · ').slice(1).join(' · ')
   423→        }
   424→        ctx.fillStyle = C.fg
   425→        ctx.fillText(txt, padL + 4, padT + 10)
   426→      }
   427→      const hovIdx = ms.inside ? clamp(Math.round(startIdx + (ms.x - padL) / barW - 0.5), 0, len - 1) : len - 1
   428→      const hcd = cs[hovIdx]
   429→      if (hcd) {
   430→        const chg = hcd.o > 0 ? ((hcd.c - hcd.o) / hcd.o) * 100 : 0
   431→        const up = hcd.c >= hcd.o
   432→        ctx.font = '10px ui-monospace, Menlo, monospace'
   433→        const y0 = padT + 34
   434→        ctx.fillStyle = C.muted
   435→        ctx.fillText('O', padL + 4, y0)
   436→        ctx.fillText('H', padL + 62, y0)
   437→        ctx.fillText('L', padL + 130, y0)
   438→        ctx.fillStyle = up ? C.pos : C.neg
   439→        ctx.fillText(fmtChartPrice(hcd.o), padL + 14, y0)
   440→        ctx.fillText(fmtChartPrice(hcd.h), padL + 72, y0)
   441→        ctx.fillText(fmtChartPrice(hcd.l), padL + 140, y0)
   442→        ctx.fillText(`C ${fmtChartPrice(hcd.c)}  ${chg >= 0 ? '+' : ''}${chg.toFixed(2)}%`, padL + 200, y0)
   443→      }
   444→    }
   445→
   446→    // магнит (Ctrl): индикатор + привязка курсора к OHLC
   447→    if (magnetRef.current && ms.inside) {
   448→      const sn = snapPoint(tOfX(ms.x), pOfY(ms.y))
   449→      const sx = xOfT(sn.t)
   450→      const sy = yOfP(sn.p)

   659→  const hitTest = useCallback((x: number, y: number): string | null => {
   660→    if (!chartKey) return null
   661→    const sc = computeScale()
   662→    if (!sc) return null
   663→    const { padL, plotW, padT, fullH } = sc

   665→    const near = (ax: number, ay: number, bx: number, by: number): boolean => {
   666→      const dx = bx - ax, dy = by - ay
   667→      const l2 = dx * dx + dy * dy
   668→      if (l2 === 0) return Math.hypot(x - ax, y - ay) < 7
   669→      let t = ((x - ax) * dx + (y - ay) * dy) / l2
   670→      t = clamp(t, 0, 1)
   671→      return Math.hypot(x - (ax + t * dx), y - (ay + t * dy)) < 7
   672→    }
   673→
   674→    for (const d of drawingsRef.current) {
   675→      const pts = d.points.map((pt) => ({ x: sc.xOfT(pt.t), y: sc.yOfP(pt.p) }))
   676→      if (d.type === 'trend' && pts.length >= 2) {
   677→        if (near(pts[0].x, pts[0].y, pts[1].x, pts[1].y)) return d.id
   678→      } else if (d.type === 'hline' && pts.length >= 1) {
   679→        if (near(padL, pts[0].y, padL + plotW, pts[0].y)) return d.id
   680→      } else if (d.type === 'vline' && pts.length >= 1) {
   681→        if (near(pts[0].x, padT, pts[0].x, padT + fullH)) return d.id
   682→      } else if (d.type === 'rect' && pts.length >= 2) {
   683→        const x1 = Math.min(pts[0].x, pts[1].x), x2 = Math.max(pts[0].x, pts[1].x)
   684→        const y1 = Math.min(pts[0].y, pts[1].y), y2 = Math.max(pts[0].y, pts[1].y)
   685→        if (near(x1, y1, x2, y1) || near(x1, y2, x2, y2) || near(x1, y1, x1, y2) || near(x2, y1, x2, y2)) return d.id
   686→      } else if (d.type === 'fib' && pts.length >= 2) {
   687→        if (near(pts[0].x, pts[0].y, pts[1].x, pts[1].y)) return d.id
   688→      }
   689→    }
   690→    return null

   897→        ms.startPRange = { min: sc.minP, max: sc.maxP }
   898→        cv.style.cursor = 'grabbing'
   899→      }
   900→      redraw()
   901→      return
   902→    }
   903→    if (t === 'ruler') {
   904→      if (ms.mode === 'draw2' && ms.pending) {
   905→        rulerRef.current = { t1: ms.pending.t, p1: ms.pending.p, t2: anchor.t, p2: anchor.p }
   906→        ms.mode = 'idle'
   907→        ms.pending = undefined
   908→        onToolDone?.()
   909→      } else {
   910→        ms.mode = 'draw2'
   911→        ms.pending = anchor
   912→      }
   913→      redraw()
   914→      return
   915→    }
   916→    if (t === 'hline' || t === 'vline') {
   917→      if (chartKey) {
   918→        drawId += 1
   919→        drawingsRef.current = [...drawingsRef.current, { id: `d${Date.now()}_${drawId}`, type: t, points: [anchor] }]
   920→        saveDrawings(chartKey, drawingsRef.current)
   921→      }
   922→      onToolDone?.()
   923→      redraw()
   924→      return
   925→    }
   926→    // двухточечные: тренд / прямоугольник / фибо
   927→    if (ms.mode === 'draw2' && ms.pending) {
   928→      if (chartKey) {
   929→        drawId += 1
   930→        drawingsRef.current = [...drawingsRef.current, { id: `d${Date.now()}_${drawId}`, type: t as DrawType, points: [ms.pending, anchor] }]
   931→        saveDrawings(chartKey, drawingsRef.current)
   932→      }
   933→      ms.mode = 'idle'
   934→      ms.pending = undefined
   935→      onToolDone?.()
   936→    } else {
   937→      ms.mode = 'draw2'
   938→      ms.pending = anchor
   939→    }
   940→    redraw()
   941→  }
   942→
   943→  const onMouseUp = () => {
   944→    const ms = mouseRef.current
   945→    // линейка колёсиком выключается на window.mouseup — отпустил, она исчезла
   946→    if (ms.mode === 'pan' || ms.mode === 'drag') {
   947→      ms.mode = 'idle'
   948→      ms.grab = undefined
   949→      const cv = canvasRef.current
   950→      if (cv) cv.style.cursor = 'crosshair'
   951→    }
   952→    redraw()
   953→  }
   954→
   955→  const onWheelNative = useCallback((e: WheelEvent) => {
   956→    if (!interactive) return
   957→    e.preventDefault()
   958→    const sc = computeScale()
   959→    if (!sc) return
   960→    const cv = canvasRef.current
   961→    if (!cv) return
   962→    const r = cv.getBoundingClientRect()
   963→    const px = e.clientX - r.left
   964→    const factor = e.deltaY > 0 ? 1.15 : 1 / 1.15
   965→    const view = viewRef.current
   966→    const newBars = clamp(view.bars * factor, 15, Math.max(60, sc.len + 40))
   967→    // якорь: индекс под курсором остаётся на месте
   968→    const anchorIdx = sc.startIdx + ((px - sc.padL) / sc.plotW) * view.bars
   969→    const newRight = anchorIdx + (1 - (px - sc.padL) / sc.plotW) * newBars
   970→    view.bars = newBars
   971→    view.offset = sc.len - newRight
   972→    redrawRef.current()
   973→  }, [computeScale, interactive])
   974→
   975→  // нативный non-passive wheel: зум колесом без прокрутки страницы
   976→  useEffect(() => {
   977→    const cv = canvasRef.current
   978→    if (!cv) return
   979→    cv.addEventListener('wheel', onWheelNative, { passive: false })
   980→    return () => cv.removeEventListener('wheel', onWheelNative)
   981→  }, [onWheelNative])
   982→
   983→  const onDoubleClick = () => {
   984→    viewRef.current = { bars: 130, offset: 0, pRange: null }
   985→    redraw()
   986→  }
   987→
   988→  const onLeave = () => {
   989→    const ms = mouseRef.current
   990→    ms.inside = false
   991→    if (ms.mode === 'pan' || ms.mode === 'drag') {
   992→      ms.mode = 'idle'
   993→      ms.grab = undefined
   994→      const cv = canvasRef.current
   995→      if (cv) cv.style.cursor = 'crosshair'
   996→    }
   997→    // линейку колёсика не отменяем на выходе из канваса —
   998→    // она выключится, когда колёсико будет отпущено (window mouseup)
   999→    redraw()
  1000→  }
  1001→
  1002→  return (
  1003→    <div ref={wrapRef} className={cn('relative h-full w-full', className)}>
  1004→      <canvas
  1005→        ref={canvasRef}
  1006→        className="absolute inset-0 h-full w-full"
  1007→        style={{ touchAction: 'none' }}
  1008→        onMouseMove={onMouseMove}
  1009→        onMouseDown={onMouseDown}
  1010→        onMouseUp={onMouseUp}
  1011→        onMouseLeave={onLeave}
  1012→        onDoubleClick={onDoubleClick}
  1013→        onContextMenu={(e) => e.preventDefault()}
  1014→      />
  1015→    </div>
  1016→  )
  1017→}
