    50→// ─── Загрузка символов биржи ─────────────────────────────────────────

    52→function useSymbols(ex: ExchangeId, market: MarketType | 'ALL') {
    53→  const [rows, setRows] = useState<SymbolRow[]>([])
    54→  const [loading, setLoading] = useState(true)
    55→  useEffect(() => {
    56→    let alive = true
    57→    const hub = getHub()
    58→    const load = async () => {
    59→      // первый await выводит из синхронного контекста эффекта
    60→      await Promise.resolve()
    61→      if (!alive) return
    62→      setLoading(true)
    63→      if (market === 'ALL') {
    64→        const [f, s] = await Promise.all([hub.fetchSymbols(ex, 'F'), hub.fetchSymbols(ex, 'S')])
    65→        if (!alive) return
    66→        const map = new Map<string, SymbolRow>()
    67→        for (const r of [...f, ...s]) {
    68→          const prev = map.get(r.symbol)
    69→          if (!prev || r.vol24h > prev.vol24h) map.set(r.symbol, r)
    70→        }
    71→        setRows([...map.values()].sort((a, b) => b.vol24h - a.vol24h))
    72→      } else {
    73→        setRows(await hub.fetchSymbols(ex, market))
    74→      }
    75→      if (alive) setLoading(false)
    76→    }
    77→    void load()
    78→    return () => { alive = false }
    79→  }, [ex, market])
    80→  return { rows, loading }
    81→}

    85→function TickerCombobox({ rows, loading, onSelect, placeholder, disabled }: {

   294→function VolumeTiersSection({ win }: { win: DensityWindowConfig }) {
   295→  const updateDensityWindow = useApp((s) => s.updateDensityWindow)
   296→  const tiers = win.volTiers ?? DEFAULT_VOL_TIERS
   297→  const setTiers = (next: VolumeTier[]) => updateDensityWindow(win.id, { volTiers: next })
   298→
   299→  return (
   300→    <div className="space-y-3 rounded-2xl border border-[var(--border)] bg-[color-mix(in_srgb,var(--muted)_60%,transparent)] p-4">
   301→      <div className="flex items-center justify-between">
   302→        <span className="flex items-center gap-2 text-[13px] font-black uppercase tracking-wider text-[var(--foreground)]">
   303→          <Palette className="h-4 w-4" style={{ color: 'var(--acc)' }} />
   304→          Цвет объёма заявки
   305→        </span>
   306→        <Button
   307→          variant="outline" size="sm"
   308→          disabled={tiers.length >= 4}
   309→          onClick={() => setTiers([...tiers, { minUsd: 20e6, color: '#22d3ee' }])}
   310→          className="h-8 rounded-xl px-2.5 text-[11px]"
   311→        >
   312→          <Plus className="mr-1 h-3 w-3" /> порог
   313→        </Button>
   314→      </div>
   315→
   316→      <div className="space-y-2">
   317→        {tiers.map((t, i) => {
   318→          const next = tiers[i + 1]
   319→          const from = t.minUsd
   320→          const to = next?.minUsd
   321→          return (
   322→            <div key={i} className="flex items-center gap-2 rounded-xl border border-[var(--border)] bg-[var(--card)] px-3 py-2">
   323→              <span className="shrink-0 text-[11px] font-semibold text-[var(--muted-foreground)]">от</span>
   324→              <div className="relative w-24">
   325→                <Input
   326→                  type="number" step={0.5} min={0} value={t.minUsd / 1e6}
   327→                  onChange={(e) => {
   328→                    const v = Math.max(0, Number(e.target.value)) * 1e6
   329→                    const next2 = [...tiers]
   330→                    next2[i] = { ...t, minUsd: v }
   331→                    setTiers(next2.sort((a, b) => a.minUsd - b.minUsd))
   332→                  }}
   333→                  className="cp-num h-9 rounded-lg border-[var(--border)] bg-[var(--muted)] pr-10 text-[13px] font-bold text-[var(--foreground)] [appearance:textfield] [&::-webkit-inner-spin-button]:appearance-none [&::-webkit-outer-spin-button]:appearance-none"

   347→                    className={cn('h-6 w-6 rounded-md border transition-transform hover:scale-110', t.color === c.value ? 'cp-glow-sm border-white/60' : 'border-transparent')}
   348→                    style={{ background: c.value }}

   353→                  onClick={() => setTiers(tiers.filter((_, j) => j !== i))}
   354→                  className="ml-1 flex h-6 w-6 items-center justify-center rounded-md text-[var(--muted-foreground)] transition-colors hover:bg-[color-mix(in_srgb,var(--destructive)_15%,transparent)] hover:text-[var(--destructive)]"

   368→        })}

   370→      <p className="text-[10px] leading-relaxed text-[var(--muted-foreground)]">
   371→        Объём лимитной заявки подсвечивается цветом по порогам — например, до $5M обычным цветом,
   372→        от $5M фиолетовым, от $10M зелёным. Действует сразу для фьючерсов и спота этого окна.
   373→      </p>
   374→    </div>
   375→  )
   376→}
