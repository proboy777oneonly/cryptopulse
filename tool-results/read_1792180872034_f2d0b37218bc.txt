    40→    40→function useSymbols(ex: ExchangeId, market: MarketType | 'ALL') {
    41→    41→  const [rows, setRows] = useState<SymbolRow[]>([])
    42→    42→  const [loading, setLoading] = useState(true)
    43→    43→  useEffect(() => {
    44→    44→    let alive = true
    45→    45→    const hub = getHub()
    46→    46→    const load = async () => {
    47→    47→      // первый await выводит из синхронного контекста эффекта
    48→    48→      await Promise.resolve()
    49→    49→      if (!alive) return
    50→    50→      setLoading(true)
    51→    51→      if (market === 'ALL') {
    52→    52→        const [f, s] = await Promise.all([hub.fetchSymbols(ex, 'F'), hub.fetchSymbols(ex, 'S')])
    53→    53→        if (!alive) return
    54→    54→        const map = new Map<string, SymbolRow>()
    55→    55→        for (const r of [...f, ...s]) {
    56→    56→          const prev = map.get(r.symbol)
    57→    57→          if (!prev || r.vol24h > prev.vol24h) map.set(r.symbol, r)
    58→    58→        }
    59→    59→        setRows([...map.values()].sort((a, b) => b.vol24h - a.vol24h))
    60→    60→      } else {
    61→    61→        setRows(await hub.fetchSymbols(ex, market))
    62→    62→      }
    63→    63→      if (alive) setLoading(false)
    64→    64→    }
    65→    65→    void load()
    66→    66→    return () => { alive = false }
    67→    67→  }, [ex, market])
    68→    68→  return { rows, loading }
    69→    69→}
