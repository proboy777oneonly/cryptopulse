    44→    44→  if (a >= 1) return p.toFixed(3)
    45→    45→  if (a >= 0.01) return p.toFixed(5)
    46→    46→  if (a >= 0.0001) return p.toFixed(7)
    47→    47→  return p.toExponential(3)
    48→    48→}

    50→    50→function fmtAxisTime(t: number, interval: string): string {
    51→    51→  const d = new Date(t)
    52→    52→  const hhmm = `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    53→    53→  if (interval === '1d' || interval === '1w') {
    54→    54→    return d.toLocaleDateString('ru-RU', { day: '2-digit', month: 'short' })
    55→    55→  }
    56→    56→  if (interval === '1h' || interval === '4h') {
    57→    57→    return `${d.toLocaleDateString('ru-RU', { day: '2-digit', month: 'short' })} ${hhmm}`
    58→    58→  }
    59→    59→  return hhmm
    60→    60→}

   165→   165→export function CandleChart({
   166→   166→  candles, interval, chartKey, tool = 'cursor', onToolDone, livePrice, chg24,
   167→   167→  theme, className, showVolume = true, interactive = true, clearTick,
   168→   168→  label, padLeft = 6,
   169→   169→}: CandleChartProps) {
   170→   170→  const wrapRef = useRef<HTMLDivElement>(null)
   171→   171→  const canvasRef = useRef<HTMLCanvasElement>(null)
   172→   172→  const viewRef = useRef<ViewBox>({ bars: 130, offset: -6, pRange: null })
   173→   173→  const magnetRef = useRef(false)
   174→   174→  const mouseRef = useRef<MouseState>({ x: 0, y: 0, inside: false, mode: 'idle', startX: 0, startY: 0, startOffset: 0 })
   175→   175→  const drawingsRef = useRef<ChartDrawing[]>([])

   177→   177→  const candlesRef = useRef<Candle[]>(candles)
   178→   178→  const toolRef = useRef(tool)
   179→   179→  const sizeRef = useRef({ w: 0, h: 0 })
   180→   180→
   181→   181→  // зеркала пропсов в рефах — только внутри эффектов
   182→   182→  useEffect(() => { candlesRef.current = candles }, [candles])
   183→   183→  useEffect(() => { toolRef.current = tool }, [tool])
   184→   184→
   185→   185→  // ── Цвета из CSS-переменных темы ──────────────────────────────────
   186→   186→  const readColors = useCallback(() => {
   187→   187→    if (typeof window === 'undefined') {
   188→   188→      return { bg: '#0b0e17', fg: '#e2e8f0', grid: 'rgba(255,255,255,0.06)', muted: '#7c8aa5', border: 'rgba(255,255,255,0.1)', pos: '#34d399', neg: '#f43f5e', acc: '#8b5cf6', card: '#111524' }
   189→   189→    }

   191→   191→    const v = (n: string, f: string) => cs.getPropertyValue(n).trim() || f
   192→   192→    return {
   193→   193→      bg: v('--card-solid', '#0b0e17'),
   194→   194→      fg: v('--foreground', '#e2e8f0'),
   195→   195→      grid: v('--border', 'rgba(255,255,255,0.1)'),
   196→   196→      muted: v('--muted-foreground', '#7c8aa5'),
   197→   197→      border: v('--border', 'rgba(255,255,255,0.1)'),
   198→   198→      pos: v('--pos', '#34d399'),
   199→   199→      neg: v('--neg', '#f43f5e'),
   200→   200→      acc: v('--acc', '#8b5cf6'),
   201→   201→      card: v('--card-solid', '#111524'),

   203→   203→  }, [])
   204→   204→
   205→   205→  // ── Геометрия ─────────────────────────────────────────────────────
   206→   206→  const geom = useCallback(() => {
   207→   207→    const w = sizeRef.current.w
   208→   208→    const h = sizeRef.current.h
   209→   209→    const padL = padLeft, padR = 64, padT = 10, padB = 20
   210→   210→    const plotW = Math.max(30, w - padL - padR)
   211→   211→    const fullH = Math.max(40, h - padT - padB)
   212→   212→    const volH = showVolume ? fullH * 0.16 : 0
   213→   213→    const chartH = fullH - volH - (showVolume ? 8 : 0)
   214→   214→    return { padL, padR, padT, padB, plotW, fullH, volH, chartH }
   215→   215→  }, [padLeft, showVolume])
   216→   216→
   217→   217→  // ── Единая шкала: видимые бары + диапазон цен ──────────────────
   218→   218→  // Единственная точка правды для отрисовки и всех обработчиков:
   219→   219→  // горизонтальный сдвиг (offset), ручная вертикальная панорама (pRange,
   220→   220→  // включается после вертикального драга) и живая цена.
   221→   221→  const computeScale = useCallback((): Scale | null => {
   222→   222→    const cs = candlesRef.current
   223→   223→    if (!cs || cs.length < 2) return null
   224→   224→    const view = viewRef.current
   225→   225→    const g = geom()
   226→   226→    const intervalMs = cs[1].t - cs[0].t || 60_000
   227→   227→    const len = cs.length
   228→   228→    view.bars = clamp(view.bars, 15, Math.max(60, len + 40))
   229→   229→    const rightIdx = clamp(len - view.offset, Math.max(5, view.bars * 0.35), len + view.bars * 0.35)
   230→   230→    const startIdx = rightIdx - view.bars
   231→   231→    const i0 = Math.max(0, Math.floor(startIdx))
   232→   232→    const i1 = Math.min(len - 1, Math.ceil(rightIdx))
   233→   233→    let minP = Infinity, maxP = -Infinity, maxV = 0
   234→   234→    for (let i = i0; i <= i1; i++) {
   235→   235→      const cd = cs[i]
   236→   236→      if (cd.l < minP) minP = cd.l
   237→   237→      if (cd.h > maxP) maxP = cd.h
   238→   238→      if (cd.v > maxV) maxV = cd.v
   239→   239→    }
   240→   240→    if (!Number.isFinite(minP) || !Number.isFinite(maxP)) {
   241→   241→      minP = 0
   242→   242→      maxP = 1
   243→   243→    }
   244→   244→    if (livePrice && livePrice > 0) {
   245→   245→      if (livePrice < minP) minP = livePrice
   246→   246→      if (livePrice > maxP) maxP = livePrice
   247→   247→    }
   248→   248→    const padP = (maxP - minP) * 0.07 || Math.abs(maxP) * 0.01 || 1
   249→   249→    let lo = minP - padP
   250→   250→    let hi = maxP + padP
   251→   251→    if (view.pRange) {
   252→   252→      lo = view.pRange.min
   253→   253→      hi = view.pRange.max
   254→   254→    }
   255→   255→    if (!(hi > lo)) hi = lo + Math.abs(lo) * 0.02 + 1
   256→   256→    const barW = g.plotW / view.bars
   257→   257→    const xOfIdx = (i: number) => g.padL + (i - startIdx + 0.5) * barW
   258→   258→    const idxOfT = (t: number) => (t - cs[0].t) / intervalMs
   259→   259→    return {
   260→   260→      cs, intervalMs, len, rightIdx, startIdx, i0, i1, minP: lo, maxP: hi, maxV, barW,
   261→   261→      padL: g.padL, padR: g.padR, padT: g.padT, padB: g.padB,
   262→   262→      plotW: g.plotW, fullH: g.fullH, volH: g.volH, chartH: g.chartH,
   263→   263→      xOfIdx, idxOfT,
   264→   264→      xOfT: (t: number) => xOfIdx(idxOfT(t)),
   265→   265→      yOfP: (p: number) => g.padT + ((hi - p) / (hi - lo)) * g.chartH,
   266→   266→      pOfY: (y: number) => hi - ((y - g.padT) / g.chartH) * (hi - lo),
   267→   267→      tOfX: (x: number) => cs[0].t + ((x - g.padL) / barW + startIdx - 0.5) * intervalMs,
   268→   268→    }
   269→   269→  }, [geom, livePrice])
   270→   270→
   271→   271→  // магнит (Ctrl): привязка точки к OHLC ближайшей свечи
   272→   272→  const snapPoint = useCallback((t: number, p: number): { t: number; p: number } => {
   273→   273→    const cs = candlesRef.current
   274→   274→    if (!magnetRef.current || !cs || cs.length < 2) return { t, p }
   275→   275→    const intervalMs = cs[1].t - cs[0].t || 60_000
   276→   276→    const idx = clamp(Math.round((t - cs[0].t) / intervalMs), 0, cs.length - 1)
   277→   277→    const cd = cs[idx]
   278→   278→    let best = cd.c
   279→   279→    for (const v of [cd.o, cd.h, cd.l, cd.c]) {
   280→   280→      if (Math.abs(v - p) < Math.abs(best - p)) best = v
   281→   281→    }
   282→   282→    return { t: cd.t, p: best }
   283→   283→  }, [])
   284→   284→
   285→   285→  // ── Главная отрисовка ─────────────────────────────────────────────
   286→   286→  const redraw = useCallback(() => {
   287→   287→    const cv = canvasRef.current
   288→   288→    if (!cv) return
   289→   289→    const ctx = cv.getContext('2d')
   290→   290→    if (!ctx) return
   291→   291→    const dpr = window.devicePixelRatio || 1
   292→   292→    const w = sizeRef.current.w
   293→   293→    const h = sizeRef.current.h
   294→   294→    if (cv.width !== Math.round(w * dpr) || cv.height !== Math.round(h * dpr)) {
   295→   295→      cv.width = Math.round(w * dpr)
   296→   296→      cv.height = Math.round(h * dpr)
   297→   297→    }
   298→   298→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   299→   299→    const C = readColors()
   300→   300→    ctx.clearRect(0, 0, w, h)
   301→   301→    ctx.fillStyle = C.bg
   302→   302→    ctx.fillRect(0, 0, w, h)
   303→   303→
   304→   304→    const sc = computeScale()
   305→   305→    if (!sc) {
   306→   306→      ctx.fillStyle = C.muted
   307→   307→      ctx.font = '12px ui-monospace, Menlo, monospace'
   308→   308→      ctx.textAlign = 'center'
   309→   309→      ctx.fillText('Загрузка свечей…', w / 2, h / 2)
   310→   310→      return
   311→   311→    }
   312→   312→    const cs = sc.cs
   313→   313→    const {
   314→   314→      padL, padR, padT, plotW, fullH, volH, chartH, barW,
   315→   315→      i0, i1, minP, maxP, maxV, len, intervalMs, startIdx,
   316→   316→    } = sc
   317→   317→    const xOfIdx = sc.xOfIdx
   318→   318→    const xOfT = sc.xOfT
   319→   319→    const yOfP = sc.yOfP
   320→   320→    const pOfY = sc.pOfY
   321→   321→    const tOfX = sc.tOfX
   322→   322→
   323→   323→    // сетка + шкала цены
   324→   324→    ctx.font = '10px ui-monospace, Menlo, monospace'
   325→   325→    ctx.textBaseline = 'middle'
   326→   326→    const step = niceStep(maxP - minP, 6)
   327→   327→    ctx.strokeStyle = C.grid
   328→   328→    ctx.lineWidth = 1
   329→   329→    ctx.fillStyle = C.muted
   330→   330→    ctx.textAlign = 'left'
   331→   331→    for (let p = Math.ceil(minP / step) * step; p <= maxP; p += step) {
   332→   332→      const y = yOfP(p)
   333→   333→      ctx.beginPath()
   334→   334→      ctx.moveTo(padL, Math.round(y) + 0.5)
   335→   335→      ctx.lineTo(padL + plotW, Math.round(y) + 0.5)
   336→   336→      ctx.stroke()
   337→   337→      const ap = fmtChartPrice(p)
   338→   338→      if (ctx.measureText(ap).width <= padR - 4) ctx.fillText(ap, padL + plotW + 6, y)
   339→   339→    }
   340→   340→
   341→   341→    // шкала времени (смена дня/месяца — яркая метка с датой)
   342→   342→    const labelEvery = Math.max(1, Math.ceil(92 / barW))
   343→   343→    ctx.textAlign = 'center'
   344→   344→    for (let i = i0; i <= i1; i++) {
   345→   345→      if (i % labelEvery !== 0) continue
   346→   346→      const x = xOfIdx(i)
   347→   347→      const d = new Date(cs[i].t)
   348→   348→      const prev = i > 0 ? new Date(cs[i - 1].t) : null
   349→   349→      const longTf = interval === '1d' || interval === '1w'
   350→   350→      const boundary = prev
   351→   351→        ? (longTf ? prev.getMonth() !== d.getMonth() : prev.getDate() !== d.getDate())
   352→   352→        : false
   353→   353→      ctx.strokeStyle = boundary ? `${C.muted}44` : C.grid
   354→   354→      ctx.lineWidth = 1
   355→   355→      ctx.beginPath()
   356→   356→      ctx.moveTo(Math.round(x) + 0.5, padT)
   357→   357→      ctx.lineTo(Math.round(x) + 0.5, padT + fullH)
   358→   358→      ctx.stroke()
   359→   359→      ctx.fillStyle = boundary ? C.fg : C.muted
   360→   360→      ctx.font = boundary
   361→   361→        ? 'bold 10px ui-monospace, Menlo, monospace'
   362→   362→        : '10px ui-monospace, Menlo, monospace'
   363→   363→      ctx.fillText(
   364→   364→        boundary
   365→   365→          ? d.toLocaleDateString('ru-RU', longTf ? { month: 'short', year: '2-digit' } : { day: '2-digit', month: 'short' })
   366→   366→          : fmtAxisTime(cs[i].t, interval),
   367→   367→        x, padT + fullH + 11,
   368→   368→      )
   369→   369→    }
   370→   370→
   371→   371→    // водяной знак тикера по центру
   372→   372→    if (label && plotW > 300) {
   373→   373→      const wm = label.split(' · ')[2] ?? label
   374→   374→      ctx.save()
   375→   375→      ctx.globalAlpha = 0.055
   376→   376→      ctx.font = `bold ${Math.round(clamp(plotW / 9, 28, 64))}px ui-monospace, Menlo, monospace`
   377→   377→      ctx.fillStyle = C.fg
   378→   378→      ctx.textAlign = 'center'
   379→   379→      ctx.fillText(wm, padL + plotW / 2, padT + chartH / 2 + 12)
   380→   380→      ctx.restore()
   381→   381→    }
   382→   382→
   383→   383→    // объёмы — с вертикальным градиентом
   384→   384→    if (showVolume && maxV > 0) {
   385→   385→      const volTop = padT + chartH + 8
   386→   386→      const gUp = ctx.createLinearGradient(0, volTop, 0, volTop + volH)
   387→   387→      gUp.addColorStop(0, `${C.pos}59`)
   388→   388→      gUp.addColorStop(1, `${C.pos}12`)
   389→   389→      const gDn = ctx.createLinearGradient(0, volTop, 0, volTop + volH)
   390→   390→      gDn.addColorStop(0, `${C.neg}59`)
   391→   391→      gDn.addColorStop(1, `${C.neg}12`)
   392→   392→      for (let i = i0; i <= i1; i++) {
   393→   393→        const cd = cs[i]
   394→   394→        const x = xOfIdx(i)
   395→   395→        const vh = (cd.v / maxV) * volH
   396→   396→        ctx.fillStyle = cd.c >= cd.o ? gUp : gDn
   397→   397→        ctx.fillRect(Math.round(x - barW * 0.33), volTop + volH - vh, Math.max(1, Math.round(barW * 0.66)), vh)
   398→   398→      }
   399→   399→    }
   400→   400→
   401→   401→    // свечи — хвост в пиксельной сетке, тело с капом ширины
   402→   402→    const bodyW = clamp(barW * 0.66, 1, 21)
   403→   403→    for (let i = i0; i <= i1; i++) {
   404→   404→      const cd = cs[i]
   405→   405→      const x = xOfIdx(i)
   406→   406→      const up = cd.c >= cd.o
   407→   407→      const col = up ? C.pos : C.neg
   408→   408→      const wx = Math.round(x) + 0.5
   409→   409→      ctx.strokeStyle = col
   410→   410→      ctx.lineWidth = 1
   411→   411→      ctx.beginPath()
   412→   412→      ctx.moveTo(wx, yOfP(cd.h))
   413→   413→      ctx.lineTo(wx, yOfP(cd.l))
   414→   414→      ctx.stroke()
   415→   415→      const yO = yOfP(cd.o)
   416→   416→      const yC = yOfP(cd.c)
   417→   417→      const top = Math.min(yO, yC)
   418→   418→      const bh = Math.max(1, Math.abs(yC - yO))
   419→   419→      ctx.fillStyle = col
   420→   420→      ctx.fillRect(Math.round(x - bodyW / 2), Math.round(top), Math.round(bodyW), Math.max(1, Math.round(bh)))
   421→   421→    }
   422→   422→
   423→   423→    // метки hi/lo видимого диапазона
   424→   424→    if (i1 > i0 + 2) {
   425→   425→      let hiI = i0
   426→   426→      let loI = i0
   427→   427→      for (let i = i0 + 1; i <= i1; i++) {
   428→   428→        if (cs[i].h > cs[hiI].h) hiI = i
   429→   429→        if (cs[i].l < cs[loI].l) loI = i
   430→   430→      }
