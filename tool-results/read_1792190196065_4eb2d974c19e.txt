    26→function clamp(v: number, a: number, b: number): number {
    27→  return Math.max(a, Math.min(b, v))
    28→}
    29→
    30→function niceStep(range: number, n: number): number {
    31→  const raw = range / Math.max(1, n)
    32→  if (!Number.isFinite(raw) || raw <= 0) return 1
    33→  const mag = Math.pow(10, Math.floor(Math.log10(raw)))
    34→  const norm = raw / mag
    35→  const step = norm < 1.5 ? 1 : norm < 3 ? 2 : norm < 7 ? 5 : 10
    36→  return step * mag
    37→}

    39→export function fmtChartPrice(p: number): string {
    40→  const a = Math.abs(p)
    41→  if (a >= 10000) return p.toLocaleString('en-US', { maximumFractionDigits: 1 })
    42→  if (a >= 100) return p.toFixed(2)
    43→  if (a >= 1) return p.toFixed(3)
    44→  if (a >= 0.01) return p.toFixed(5)
    45→  if (a >= 0.0001) return p.toFixed(7)
    46→  return p.toExponential(3)
    47→}

    49→function fmtAxisTime(t: number, interval: string): string {
    50→  const d = new Date(t)
    51→  const hhmm = `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    52→  if (interval === '1d' || interval === '1w') {
    53→    return d.toLocaleDateString('ru-RU', { day: '2-digit', month: 'short' })
    54→  }
    55→  if (interval === '1h' || interval === '4h') {
    56→    return `${d.toLocaleDateString('ru-RU', { day: '2-digit', month: 'short' })} ${hhmm}`
    57→  }
    58→  return hhmm
    59→}
    60→
    61→function fmtDuration(ms: number): string {
    62→  const s = Math.round(ms / 1000)
    63→  if (s < 90) return `${s}с`
    64→  const m = Math.round(s / 60)
    65→  if (m < 90) return `${m}м`
    66→  const h = Math.round(m / 60)
    67→  if (h < 36) return `${h}ч`
    68→  return `${Math.round(h / 24)}д`
    69→}

    86→function loadDrawings(key: string): ChartDrawing[] {
    87→  try {
    88→    const raw = localStorage.getItem(`cp-draw-${key}`)
    89→    if (!raw) return []
    90→    const arr = JSON.parse(raw) as ChartDrawing[]
    91→    return Array.isArray(arr) ? arr.filter((d) => d && d.type && Array.isArray(d.points)) : []
    92→  } catch {
    93→    return []
    94→  }
    95→}

    99→    localStorage.setItem(`cp-draw-${key}`, JSON.stringify(list))
   100→  } catch { /* quota — игнорируем */ }
   101→}

   121→export function CandleChart({
   122→  candles, interval, chartKey, tool = 'cursor', onToolDone, livePrice,
   123→  theme, className, showVolume = true, interactive = true, clearTick,
   124→}: CandleChartProps) {
   125→  const wrapRef = useRef<HTMLDivElement>(null)
   126→  const canvasRef = useRef<HTMLCanvasElement>(null)
   127→  const viewRef = useRef({ bars: 130, offset: 0 })
   128→  const mouseRef = useRef<MouseState>({ x: 0, y: 0, inside: false, mode: 'idle', startX: 0, startY: 0, startOffset: 0 })
   129→  const drawingsRef = useRef<ChartDrawing[]>([])
   130→  const rulerRef = useRef<{ t1: number; p1: number; t2: number; p2: number } | null>(null)
   131→  const candlesRef = useRef<Candle[]>(candles)
   132→  const toolRef = useRef(tool)
   133→  const sizeRef = useRef({ w: 0, h: 0 })
   134→
   135→  // зеркала пропсов в рефах — только внутри эффектов
   136→  useEffect(() => { candlesRef.current = candles }, [candles])
   137→  useEffect(() => { toolRef.current = tool }, [tool])
   138→
   139→  // ── Цвета из CSS-переменных темы ──────────────────────────────────
   140→  const readColors = useCallback(() => {
   141→    if (typeof window === 'undefined') {
   142→      return { bg: '#0b0e17', fg: '#e2e8f0', grid: 'rgba(255,255,255,0.06)', muted: '#7c8aa5', border: 'rgba(255,255,255,0.1)', pos: '#34d399', neg: '#f43f5e', acc: '#8b5cf6', card: '#111524' }
   143→    }
   144→    const cs = getComputedStyle(document.documentElement)
   145→    const v = (n: string, f: string) => cs.getPropertyValue(n).trim() || f
   146→    return {
   147→      bg: v('--card-solid', '#0b0e17'),
   148→      fg: v('--foreground', '#e2e8f0'),
   149→      grid: v('--border', 'rgba(255,255,255,0.1)'),
   150→      muted: v('--muted-foreground', '#7c8aa5'),
   151→      border: v('--border', 'rgba(255,255,255,0.1)'),
   152→      pos: v('--pos', '#34d399'),
   153→      neg: v('--neg', '#f43f5e'),
   154→      acc: v('--acc', '#8b5cf6'),
   155→      card: v('--card-solid', '#111524'),
   156→    }
   157→  }, [])
   158→
   159→  // ── Геометрия ─────────────────────────────────────────────────────
   160→  const geom = useCallback(() => {
   161→    const w = sizeRef.current.w
   162→    const h = sizeRef.current.h
   163→    const padL = 6, padR = 64, padT = 10, padB = 20
   164→    const plotW = Math.max(30, w - padL - padR)
   165→    const fullH = Math.max(40, h - padT - padB)
   166→    const volH = showVolume ? fullH * 0.16 : 0
   167→    const chartH = fullH - volH - (showVolume ? 8 : 0)
   168→    return { padL, padR, padT, padB, plotW, fullH, volH, chartH }
   169→  }, [showVolume])
   170→
   171→  // ── Главная отрисовка ─────────────────────────────────────────────
   172→  const redraw = useCallback(() => {
   173→    const cv = canvasRef.current
   174→    if (!cv) return
   175→    const ctx = cv.getContext('2d')
   176→    if (!ctx) return
   177→    const dpr = window.devicePixelRatio || 1
   178→    const w = sizeRef.current.w
   179→    const h = sizeRef.current.h
   180→    if (cv.width !== Math.round(w * dpr) || cv.height !== Math.round(h * dpr)) {
   181→      cv.width = Math.round(w * dpr)
   182→      cv.height = Math.round(h * dpr)
   183→    }
   184→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   185→    const C = readColors()
   186→    ctx.clearRect(0, 0, w, h)
   187→    ctx.fillStyle = C.bg
   188→    ctx.fillRect(0, 0, w, h)
   189→
   190→    const cs = candlesRef.current
   191→    if (!cs || cs.length < 2) {
   192→      ctx.fillStyle = C.muted
   193→      ctx.font = '12px ui-monospace, Menlo, monospace'
   194→      ctx.textAlign = 'center'
   195→      ctx.fillText('Загрузка свечей…', w / 2, h / 2)
   196→      return
   197→    }
   198→
   199→    const intervalMs = cs[1].t - cs[0].t || 60_000
   200→    const view = viewRef.current
   201→    view.bars = clamp(view.bars, 15, Math.max(60, cs.length + 40))
   202→    const len = cs.length
   203→    let rightIdx = len - view.offset
   204→    rightIdx = clamp(rightIdx, Math.max(5, view.bars * 0.35), len + view.bars * 0.35)
   205→    const startIdx = rightIdx - view.bars
   206→    const { padL, padR, padT, plotW, volH, chartH, fullH } = geom()
   207→    const barW = plotW / view.bars
   208→    const xOfIdx = (i: number) => padL + (i - startIdx + 0.5) * barW
   209→    const idxOfT = (t: number) => (t - cs[0].t) / intervalMs
   210→    const xOfT = (t: number) => xOfIdx(idxOfT(t))
   211→
   212→    const i0 = Math.max(0, Math.floor(startIdx))
   213→    const i1 = Math.min(len - 1, Math.ceil(rightIdx))
   214→    let minP = Infinity, maxP = -Infinity, maxV = 0
   215→    for (let i = i0; i <= i1; i++) {
   216→      if (cs[i].l < minP) minP = cs[i].l
   217→      if (cs[i].h > maxP) maxP = cs[i].h
   218→      if (cs[i].v > maxV) maxV = cs[i].v
   219→    }
   220→    if (livePrice && livePrice > 0) {
   221→      if (livePrice < minP) minP = livePrice
   222→      if (livePrice > maxP) maxP = livePrice
   223→    }
   224→    const padP = (maxP - minP) * 0.07 || Math.abs(maxP) * 0.01 || 1
   225→    minP -= padP
   226→    maxP += padP
   227→    const yOfP = (p: number) => padT + ((maxP - p) / (maxP - minP)) * chartH
   228→    const pOfY = (y: number) => maxP - ((y - padT) / chartH) * (maxP - minP)
   229→    const tOfX = (x: number) => cs[0].t + ((x - padL) / barW + startIdx - 0.5) * intervalMs
   230→
   231→    // сетка + шкала цены
   232→    ctx.font = '10px ui-monospace, Menlo, monospace'
   233→    ctx.textBaseline = 'middle'
   234→    const step = niceStep(maxP - minP, 6)
   235→    ctx.strokeStyle = C.grid
   236→    ctx.lineWidth = 1
   237→    ctx.fillStyle = C.muted
   238→    ctx.textAlign = 'left'
   239→    for (let p = Math.ceil(minP / step) * step; p <= maxP; p += step) {
   240→      const y = yOfP(p)
   241→      ctx.beginPath()
   242→      ctx.moveTo(padL, Math.round(y) + 0.5)
   243→      ctx.lineTo(padL + plotW, Math.round(y) + 0.5)
   244→      ctx.stroke()
   245→      ctx.fillText(fmtChartPrice(p), padL + plotW + 6, y)
   246→    }
   247→
   248→    // шкала времени
   249→    const labelEvery = Math.max(1, Math.ceil(92 / barW))
   250→    ctx.textAlign = 'center'
   251→    for (let i = i0; i <= i1; i++) {
   252→      if (i % labelEvery !== 0) continue
   253→      const x = xOfIdx(i)
   254→      ctx.strokeStyle = C.grid
   255→      ctx.beginPath()
   256→      ctx.moveTo(Math.round(x) + 0.5, padT)
   257→      ctx.lineTo(Math.round(x) + 0.5, padT + fullH)
   258→      ctx.stroke()
   259→      ctx.fillStyle = C.muted
   260→      ctx.fillText(fmtAxisTime(cs[i].t, interval), x, padT + fullH + 11)
   261→    }
   262→
   263→    // объёмы
   264→    if (showVolume && maxV > 0) {
   265→      const volTop = padT + chartH + 8
   266→      for (let i = i0; i <= i1; i++) {
   267→        const cd = cs[i]
   268→        const x = xOfIdx(i)
   269→        const vh = (cd.v / maxV) * volH
   270→        ctx.fillStyle = cd.c >= cd.o ? `${C.pos}33` : `${C.neg}33`
   271→        ctx.fillRect(x - barW * 0.33, volTop + volH - vh, Math.max(1, barW * 0.66), vh)
   272→      }
   273→    }
   274→
   275→    // свечи
   276→    const bodyW = Math.max(1, barW * 0.66)
   277→    for (let i = i0; i <= i1; i++) {
   278→      const cd = cs[i]
   279→      const x = xOfIdx(i)
   280→      const up = cd.c >= cd.o
   281→      const col = up ? C.pos : C.neg
   282→      ctx.strokeStyle = col
   283→      ctx.lineWidth = 1
   284→      ctx.beginPath()
   285→      ctx.moveTo(Math.round(x) + 0.5, yOfP(cd.h))
   286→      ctx.lineTo(Math.round(x) + 0.5, yOfP(cd.l))
   287→      ctx.stroke()
   288→      const yO = yOfP(cd.o)
   289→      const yC = yOfP(cd.c)
   290→      const top = Math.min(yO, yC)
   291→      const bh = Math.max(1, Math.abs(yC - yO))
   292→      ctx.fillStyle = col
   293→      ctx.fillRect(x - bodyW / 2, top, bodyW, bh)
   294→    }
   295→
   296→    // последняя цена
   297→    const lastC = cs[len - 1]
   298→    const lastP = livePrice && livePrice > 0 ? livePrice : lastC.c
   299→    const lastUp = lastP >= lastC.o
   300→    const yLast = yOfP(lastP)
   301→    ctx.setLineDash([4, 4])
   302→    ctx.strokeStyle = lastUp ? C.pos : C.neg
   303→    ctx.lineWidth = 1
   304→    ctx.beginPath()
   305→    ctx.moveTo(padL, Math.round(yLast) + 0.5)
   306→    ctx.lineTo(padL + plotW, Math.round(yLast) + 0.5)
   307→    ctx.stroke()
   308→    ctx.setLineDash([])
   309→    // чип последней цены
   310→    ctx.fillStyle = lastUp ? C.pos : C.neg
   311→    const lpTxt = fmtChartPrice(lastP)
   312→    ctx.font = 'bold 10px ui-monospace, Menlo, monospace'
   313→    const lpW = ctx.measureText(lpTxt).width + 10
   314→    roundRect(ctx, padL + plotW + 2, yLast - 8, Math.min(lpW, padR - 4), 16, 4)
   315→    ctx.fill()
   316→    ctx.fillStyle = '#0b0e17'
   317→    ctx.textAlign = 'left'
   318→    ctx.fillText(lpTxt, padL + plotW + 7, yLast)
   319→
   320→    // легенда OHLC hovered
   321→    const ms = mouseRef.current
   322→    const hovIdx = ms.inside ? clamp(Math.round(startIdx + (ms.x - padL) / barW - 0.5), 0, len - 1) : len - 1
   323→    const hcd = cs[hovIdx]
   324→    if (hcd) {
   325→      const chg = hcd.o > 0 ? ((hcd.c - hcd.o) / hcd.o) * 100 : 0
   326→      const up = hcd.c >= hcd.o
   327→      ctx.font = '10px ui-monospace, Menlo, monospace'
   328→      ctx.textAlign = 'left'
   329→      ctx.fillStyle = C.muted
   330→      const y0 = padT + 2
   331→      ctx.fillText('O', padL + 4, y0)
   332→      ctx.fillText('H', padL + 62, y0)
   333→      ctx.fillText('L', padL + 130, y0)
   334→      ctx.fillStyle = up ? C.pos : C.neg
   335→      ctx.fillText(fmtChartPrice(hcd.o), padL + 14, y0)
   336→      ctx.fillText(fmtChartPrice(hcd.h), padL + 72, y0)
   337→      ctx.fillText(fmtChartPrice(hcd.l), padL + 140, y0)
   338→      ctx.fillText(`C ${fmtChartPrice(hcd.c)}  ${chg >= 0 ? '+' : ''}${chg.toFixed(2)}%`, padL + 200, y0)
   339→    }
   340→
   341→    // ── Рисунки ────────────────────────────────────────────────────
   342→    const strokeDrawing = (d: ChartDrawing, hover: boolean) => {
   343→      const pts = d.points.map((pt) => ({ x: xOfT(pt.t), y: yOfP(pt.p) }))
   344→      const col = hover ? C.acc : C.acc
   345→      ctx.strokeStyle = col
   346→      ctx.lineWidth = hover ? 2 : 1.5
   347→      ctx.setLineDash([])
   348→      if (d.type === 'trend' && pts.length >= 2) {
   349→        ctx.beginPath()
   350→        ctx.moveTo(pts[0].x, pts[0].y)
   351→        ctx.lineTo(pts[1].x, pts[1].y)
   352→        ctx.stroke()
   353→        dot(pts[0]); dot(pts[1])
   354→      } else if (d.type === 'hline' && pts.length >= 1) {
   355→        const y = pts[0].y
   356→        ctx.beginPath()
   357→        ctx.moveTo(padL, y)
   358→        ctx.lineTo(padL + plotW, y)
   359→        ctx.stroke()
   360→        // чип цены справа
   361→        ctx.fillStyle = col
   362→        ctx.font = 'bold 9px ui-monospace, Menlo, monospace'
   363→        const txt = fmtChartPrice(d.points[0].p)
   364→        const tw = ctx.measureText(txt).width + 8
   365→        roundRect(ctx, padL + plotW + 2, y - 7, Math.min(tw, padR - 4), 14, 3)
   366→        ctx.fill()
   367→        ctx.fillStyle = '#0b0e17'
   368→        ctx.fillText(txt, padL + plotW + 6, y)
   369→      } else if (d.type === 'vline' && pts.length >= 1) {
   370→        const x = pts[0].x
   371→        ctx.setLineDash([2, 3])
   372→        ctx.beginPath()
   373→        ctx.moveTo(x, padT)
   374→        ctx.lineTo(x, padT + fullH)
   375→        ctx.stroke()
   376→        ctx.setLineDash([])
   377→      } else if (d.type === 'rect' && pts.length >= 2) {
   378→        const x = Math.min(pts[0].x, pts[1].x), y = Math.min(pts[0].y, pts[1].y)
   379→        const rw = Math.abs(pts[1].x - pts[0].x), rh = Math.abs(pts[1].y - pts[0].y)
   380→        ctx.fillStyle = `${C.acc}14`
   381→        ctx.fillRect(x, y, rw, rh)
   382→        ctx.strokeRect(x, y, rw, rh)
   383→        dot(pts[0]); dot(pts[1])
   384→      } else if (d.type === 'fib' && pts.length >= 2) {
   385→        const p1 = d.points[0], p2 = d.points[1]
   386→        const xa = Math.min(pts[0].x, pts[1].x), xb = Math.max(pts[0].x, pts[1].x)
   387→        ctx.strokeStyle = `${C.acc}cc`
   388→        ctx.beginPath()
   389→        ctx.moveTo(pts[0].x, pts[0].y)
   390→        ctx.lineTo(pts[1].x, pts[1].y)
   391→        ctx.stroke()
   392→        ctx.font = '9px ui-monospace, Menlo, monospace'
   393→        ctx.textAlign = 'left'
   394→        for (const lv of FIB_LEVELS) {
   395→          const price = p1.p + (p2.p - p1.p) * lv
   396→          const y = yOfP(price)
   397→          ctx.strokeStyle = lv === 0 || lv === 1 ? `${C.acc}aa` : `${C.acc}55`
   398→          ctx.setLineDash(lv === 0.5 ? [3, 3] : [])
   399→          ctx.beginPath()
   400→          ctx.moveTo(xa, y)
   401→          ctx.lineTo(xb + 26, y)
   402→          ctx.stroke()
   403→          ctx.setLineDash([])
   404→          ctx.fillStyle = `${C.acc}dd`
   405→          ctx.fillText(`${lv.toFixed(3)}  ${fmtChartPrice(price)}`, xa + 4, y - 6)
   406→        }
   407→        dot(pts[0]); dot(pts[1])
   408→      }
   409→    }
   410→    const dot = (pt: { x: number; y: number }) => {
   411→      ctx.fillStyle = C.acc
   412→      ctx.beginPath()
   413→      ctx.arc(pt.x, pt.y, 2.5, 0, Math.PI * 2)
   414→      ctx.fill()
   415→    }
   416→    function roundRect(c: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   417→      c.beginPath()
   418→      c.moveTo(x + r, y)
   419→      c.arcTo(x + w, y, x + w, y + h, r)
   420→      c.arcTo(x + w, y + h, x, y + h, r)
   421→      c.arcTo(x, y + h, x, y, r)
   422→      c.arcTo(x, y, x + w, y, r)
   423→      c.closePath()
   424→    }
   425→
   426→    for (const d of drawingsRef.current) {
   427→      strokeDrawing(d, ms.hoverId === d.id)
   428→    }
   429→
   430→    // предпросмотр рисования
   431→    if (ms.mode === 'draw2' && ms.pending) {
   432→      const p2 = { t: tOfX(ms.x), p: pOfY(ms.y) }
   433→      const preview: ChartDrawing = {
   434→        id: 'preview',
   435→        type: (TWO_POINT.includes(toolRef.current as DrawType) ? toolRef.current : 'trend') as DrawType,
   436→        points: [ms.pending, p2],
   437→      }
   438→      ctx.globalAlpha = 0.7
   439→      strokeDrawing(preview, false)
   440→      ctx.globalAlpha = 1
   441→    }
   442→
   443→    // ── Линейка ────────────────────────────────────────────────────
   444→    const r = rulerRef.current
   445→    const rulerLive = ms.mode === 'ruler' && ms.pending

   533→    const i0 = Math.max(0, Math.floor(startIdx)), i1 = Math.min(len - 1, Math.ceil(rightIdx))
   534→    for (let i = i0; i <= i1; i++) {
   535→      if (cs[i].l < minP) minP = cs[i].l
   536→      if (cs[i].h > maxP) maxP = cs[i].h
   537→    }

   540→    const yOfP = (p: number) => 10 + ((maxP - p) / (maxP - minP)) * chartH
   541→    const xOfT = (t: number) => padL + ((t - cs[0].t) / intervalMs - startIdx + 0.5) * barW

   544→      const dx = bx - ax, dy = by - ay
   545→      const l2 = dx * dx + dy * dy
   546→      if (l2 === 0) return Math.hypot(x - ax, y - ay) < 7
   547→      let t = ((x - ax) * dx + (y - ay) * dy) / l2
   548→      t = clamp(t, 0, 1)
   549→      return Math.hypot(x - (ax + t * dx), y - (ay + t * dy)) < 7
   550→    }
   551→
   552→    for (const d of drawingsRef.current) {
   553→      const pts = d.points.map((pt) => ({ x: xOfT(pt.t), y: yOfP(pt.p) }))
   554→      if (d.type === 'trend' && pts.length >= 2) {
   555→        if (near(pts[0].x, pts[0].y, pts[1].x, pts[1].y)) return d.id
   556→      } else if (d.type === 'hline' && pts.length >= 1) {
   557→        if (near(padL, pts[0].y, padL + plotW, pts[0].y)) return d.id
   558→      } else if (d.type === 'vline' && pts.length >= 1) {
   559→        if (near(pts[0].x, 10, pts[0].x, 10 + fullH)) return d.id
   560→      } else if (d.type === 'rect' && pts.length >= 2) {
   561→        const x1 = Math.min(pts[0].x, pts[1].x), x2 = Math.max(pts[0].x, pts[1].x)
   562→        const y1 = Math.min(pts[0].y, pts[1].y), y2 = Math.max(pts[0].y, pts[1].y)
   563→        if (near(x1, y1, x2, y1) || near(x1, y2, x2, y2) || near(x1, y1, x1, y2) || near(x2, y1, x2, y2)) return d.id

   565→        if (near(pts[0].x, pts[0].y, pts[1].x, pts[1].y)) return d.id
   566→      }
   567→    }

   880→    viewRef.current = { bars: 130, offset: 0 }
   881→    redraw()
   882→  }
   883→
   884→  const onLeave = () => {
   885→    const ms = mouseRef.current
   886→    ms.inside = false
   887→    // драг/линейка завершились вне канваса — фиксируем
   888→    if (ms.mode === 'ruler' && ms.pending) {
   889→      rulerRef.current = { t1: ms.pending.t, p1: ms.pending.p, t2: tOfXFromState(ms.x), p2: pOfYFromState(ms.y) }
   890→      ms.pending = undefined
   891→      ms.mode = 'idle'
   892→    } else if (ms.mode === 'pan' || ms.mode === 'drag') {
   893→      ms.mode = 'idle'
   894→      ms.grab = undefined
   895→      const cv = canvasRef.current
   896→      if (cv) cv.style.cursor = 'crosshair'
   897→    }
   898→    redraw()
   899→  }
   900→
   901→  return (
   902→    <div ref={wrapRef} className={cn('relative h-full w-full', className)}>
   903→      <canvas
   904→        ref={canvasRef}
   905→        className="absolute inset-0 h-full w-full"
   906→        style={{ touchAction: 'none' }}
   907→        onMouseMove={onMouseMove}
   908→        onMouseDown={onMouseDown}
   909→        onMouseUp={onMouseUp}
   910→        onMouseLeave={onLeave}
   911→        onDoubleClick={onDoubleClick}
   912→        onContextMenu={(e) => e.preventDefault()}
   913→      />
   914→    </div>
   915→  )
   916→}
