    50→// ─── Загрузка символов биржи ─────────────────────────────────────────

    52→function useSymbols(ex: ExchangeId, market: MarketType | 'ALL') {
    53→  const [rows, setRows] = useState<SymbolRow[]>([])
    54→  const [loading, setLoading] = useState(true)
    55→  useEffect(() => {
    56→    let alive = true
    57→    const hub = getHub()
    58→    const load = async () => {
    59→      // первый await выводит из синхронного контекста эффекта
    60→      await Promise.resolve()
    61→      if (!alive) return
    62→      setLoading(true)
    63→      if (market === 'ALL') {
    64→        const [f, s] = await Promise.all([hub.fetchSymbols(ex, 'F'), hub.fetchSymbols(ex, 'S')])
    65→        if (!alive) return
    66→        const map = new Map<string, SymbolRow>()
    67→        for (const r of [...f, ...s]) {
    68→          const prev = map.get(r.symbol)
    69→          if (!prev || r.vol24h > prev.vol24h) map.set(r.symbol, r)
    70→        }
    71→        setRows([...map.values()].sort((a, b) => b.vol24h - a.vol24h))
    72→      } else {
    73→        setRows(await hub.fetchSymbols(ex, market))
    74→      }
    75→      if (alive) setLoading(false)
    76→    }
    77→    void load()
    78→    return () => { alive = false }
    79→  }, [ex, market])
    80→  return { rows, loading }
    81→}

    85→function TickerCombobox({ rows, loading, onSelect, placeholder, disabled }: {

   294→function VolumeTiersSection({ win }: { win: DensityWindowConfig }) {

   296→  const tiers = win.volTiers ?? DEFAULT_VOL_TIERS

   301→      <div className="flex items-center justify-between">

   303→          <Palette className="h-4 w-4" style={{ color: 'var(--acc)' }} />
   304→          Цвет объёма заявки
   305→        </span>
   306→        <Button
   307→          variant="outline" size="sm"
   308→          disabled={tiers.length >= 4}
   309→          onClick={() => setTiers([...tiers, { minUsd: 20e6, color: '#22d3ee' }])}
   310→          className="h-8 rounded-xl px-2.5 text-[11px]"
   311→        >
   312→          <Plus className="mr-1 h-3 w-3" /> порог
   313→        </Button>
   314→      </div>
   315→
   316→      <div className="space-y-2">
   317→        {tiers.map((t, i) => {
   318→          const next = tiers[i + 1]
   319→          const from = t.minUsd
   320→          const to = next?.minUsd
   321→          return (
   322→            <div key={i} className="flex items-center gap-2 rounded-xl border border-[var(--border)] bg-[var(--card)] px-3 py-2">
   323→              <span className="shrink-0 text-[11px] font-semibold text-[var(--muted-foreground)]">от</span>

   326→                  type="number" step={0.5} min={0} value={t.minUsd / 1e6}

   375→      </p>
   376→    </div>
   377→  )
   378→}

   806→export function DensitiesView() {
   807→  const densityWindows = useApp((s) => s.densityWindows)
   808→  const addDensityWindow = useApp((s) => s.addDensityWindow)
   809→  const sendAllTickers = useApp((s) => s.sendAllTickers)
   810→  const compact = useApp((s) => s.effects.compact)
   811→  const densityEvents = useLive((s) => s.densityEvents)
   812→  const dataMode = useLive((s) => s.dataMode)
   813→  const uptime = useUptime()
   814→
   815→  const [addOpen, setAddOpen] = useState(false)
   816→  const [addEx, setAddEx] = useState('')
   817→
   818→  const totalActive = useMemo(
   819→    () => Object.values(densityEvents).reduce((acc, list) => acc + list.length, 0),
   820→    [densityEvents],
   821→  )
   822→  const unused = useMemo(
   823→    () => Object.values(EX_MAP).filter((e) => !densityWindows.some((w) => w.exchange === e.id)),
   824→    [densityWindows],
   825→  )
   826→
   827→  return (
   828→    <div className="space-y-4">
   829→      {/* Тулбар */}
   830→      <div className="cp-border-glow flex flex-wrap items-center gap-2.5 rounded-2xl bg-card px-4 py-3">
   831→        <div className="flex items-center gap-2 text-sm font-black">
   832→          <Layers className="h-4 w-4" style={{ color: 'var(--acc)' }} />
   833→          {densityWindows.length} окон · 1 окно = 1 биржа
   834→        </div>
   835→        <div className="cp-num flex items-center gap-2 rounded-xl border border-[var(--border)] bg-[var(--muted)] px-3 py-1.5 text-xs">
   836→          <span className={cn('h-1.5 w-1.5 rounded-full', dataMode === 'live' && 'cp-pulse-dot')} style={{ background: dataMode === 'live' ? 'var(--pos)' : '#f59e0b' }} />
   837→          <span className="font-bold" style={{ color: 'var(--acc)' }}>{totalActive} активных</span>
   838→        </div>
   839→        <div className="cp-num flex items-center gap-1.5 rounded-xl border border-[var(--border)] bg-[var(--muted)] px-3 py-1.5 text-xs font-bold">
   840→          <Timer className="h-3.5 w-3.5 text-[var(--muted-foreground)]" /> {uptime}
   841→        </div>
   842→        <div className="ml-auto flex items-center gap-2">
   843→          <Button
   844→            variant="outline" size="sm"
   845→            onClick={() => {
   846→              densityWindows.forEach((w) => sendAllTickers(w.id))
   847→              toastSuccess('Синхронизация тикеров', 'Все тикеры отправлены во все окна')
   848→            }}
   849→            title="Каждое окно отправляет свои тикеры во все остальные"
   850→          >
   851→            <Send className="mr-1.5 h-3.5 w-3.5" /> Все тикеры → все окна
   852→          </Button>
   853→          <Button size="sm" disabled={unused.length === 0 || densityWindows.length >= 8} onClick={() => setAddOpen(true)}>
   854→            <Plus className="mr-1.5 h-4 w-4" /> Добавить окно
   855→          </Button>
   856→        </div>
   857→      </div>
   858→
   859→      {/* Сетка окон */}
   860→      <div className="grid gap-4 md:grid-cols-2 2xl:grid-cols-3">
   861→        {densityWindows.map((w) => (
   862→          <DensityWindowCard key={w.id} win={w} compact={compact} />
   863→        ))}
   864→      </div>
   865→
   866→      {/* Диалог добавления окна */}
   867→      <Dialog open={addOpen} onOpenChange={(v) => { setAddOpen(v); if (!v) setAddEx('') }}>
   868→        <DialogContent className="max-w-xs rounded-3xl border-[var(--border)] bg-[var(--card-solid)]">
   869→          <DialogHeader><DialogTitle className="text-base">Новое окно плотностей</DialogTitle></DialogHeader>
   870→          <Select value={addEx} onValueChange={setAddEx}>
   871→            <SelectTrigger><SelectValue placeholder="Биржа" /></SelectTrigger>
   872→            <SelectContent>
   873→              {unused.map((e) => (
   874→                <SelectItem key={e.id} value={e.id}>
   875→                  <span className="flex items-center gap-2"><span className="h-2 w-2 rounded-full" style={{ background: e.color }} />{e.name}</span>
   876→                </SelectItem>
   877→              ))}
   878→            </SelectContent>
   879→          </Select>
   880→          <Button disabled={!addEx} onClick={() => { addDensityWindow(addEx as ExchangeId); setAddOpen(false); setAddEx(''); toastSuccess('Окно создано', `${EX_MAP[addEx as ExchangeId].name} подключено`) }}>
   881→            <Plus className="mr-2 h-4 w-4" /> Создать окно
   882→          </Button>
   883→        </DialogContent>
   884→      </Dialog>
   885→    </div>
   886→  )
   887→}
