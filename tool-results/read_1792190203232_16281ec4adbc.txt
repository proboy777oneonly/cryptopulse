    30→    30→function niceStep(range: number, n: number): number {
    31→    31→  const raw = range / Math.max(1, n)
    32→    32→  if (!Number.isFinite(raw) || raw <= 0) return 1
    33→    33→  const mag = Math.pow(10, Math.floor(Math.log10(raw)))
    34→    34→  const norm = raw / mag
    35→    35→  const step = norm < 1.5 ? 1 : norm < 3 ? 2 : norm < 7 ? 5 : 10
    36→    36→  return step * mag
    37→    37→}

    39→    39→export function fmtChartPrice(p: number): string {
    40→    40→  const a = Math.abs(p)
    41→    41→  if (a >= 10000) return p.toLocaleString('en-US', { maximumFractionDigits: 1 })
    42→    42→  if (a >= 100) return p.toFixed(2)
    43→    43→  if (a >= 1) return p.toFixed(3)
    44→    44→  if (a >= 0.01) return p.toFixed(5)
    45→    45→  if (a >= 0.0001) return p.toFixed(7)
    46→    46→  return p.toExponential(3)
    47→    47→}

    49→    49→function fmtAxisTime(t: number, interval: string): string {
    50→    50→  const d = new Date(t)
    51→    51→  const hhmm = `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
    52→    52→  if (interval === '1d' || interval === '1w') {
    53→    53→    return d.toLocaleDateString('ru-RU', { day: '2-digit', month: 'short' })
    54→    54→  }
    55→    55→  if (interval === '1h' || interval === '4h') {
    56→    56→    return `${d.toLocaleDateString('ru-RU', { day: '2-digit', month: 'short' })} ${hhmm}`
    57→    57→  }
    58→    58→  return hhmm
    59→    59→}

    61→    61→function fmtDuration(ms: number): string {
    62→    62→  const s = Math.round(ms / 1000)
    63→    63→  if (s < 90) return `${s}с`
    64→    64→  const m = Math.round(s / 60)
    65→    65→  if (m < 90) return `${m}м`
    66→    66→  const h = Math.round(m / 60)
    67→    67→  if (h < 36) return `${h}ч`
    68→    68→  return `${Math.round(h / 24)}д`
    69→    69→}

   121→   121→export function CandleChart({
   122→   122→  candles, interval, chartKey, tool = 'cursor', onToolDone, livePrice,
   123→   123→  theme, className, showVolume = true, interactive = true, clearTick,
   124→   124→}: CandleChartProps) {
   125→   125→  const wrapRef = useRef<HTMLDivElement>(null)
   126→   126→  const canvasRef = useRef<HTMLCanvasElement>(null)
   127→   127→  const viewRef = useRef({ bars: 130, offset: 0 })
   128→   128→  const mouseRef = useRef<MouseState>({ x: 0, y: 0, inside: false, mode: 'idle', startX: 0, startY: 0, startOffset: 0 })
   129→   129→  const drawingsRef = useRef<ChartDrawing[]>([])
   130→   130→  const rulerRef = useRef<{ t1: number; p1: number; t2: number; p2: number } | null>(null)
   131→   131→  const candlesRef = useRef<Candle[]>(candles)
   132→   132→  const toolRef = useRef(tool)
   133→   133→  const sizeRef = useRef({ w: 0, h: 0 })
   134→   134→
   135→   135→  // зеркала пропсов в рефах — только внутри эффектов
   136→   136→  useEffect(() => { candlesRef.current = candles }, [candles])
   137→   137→  useEffect(() => { toolRef.current = tool }, [tool])
   138→   138→
   139→   139→  // ── Цвета из CSS-переменных темы ──────────────────────────────────
   140→   140→  const readColors = useCallback(() => {
   141→   141→    if (typeof window === 'undefined') {
   142→   142→      return { bg: '#0b0e17', fg: '#e2e8f0', grid: 'rgba(255,255,255,0.06)', muted: '#7c8aa5', border: 'rgba(255,255,255,0.1)', pos: '#34d399', neg: '#f43f5e', acc: '#8b5cf6', card: '#111524' }
   143→   143→    }
   144→   144→    const cs = getComputedStyle(document.documentElement)
   145→   145→    const v = (n: string, f: string) => cs.getPropertyValue(n).trim() || f
   146→   146→    return {
   147→   147→      bg: v('--card-solid', '#0b0e17'),
   148→   148→      fg: v('--foreground', '#e2e8f0'),
   149→   149→      grid: v('--border', 'rgba(255,255,255,0.1)'),
   150→   150→      muted: v('--muted-foreground', '#7c8aa5'),
   151→   151→      border: v('--border', 'rgba(255,255,255,0.1)'),
   152→   152→      pos: v('--pos', '#34d399'),
   153→   153→      neg: v('--neg', '#f43f5e'),
   154→   154→      acc: v('--acc', '#8b5cf6'),
   155→   155→      card: v('--card-solid', '#111524'),
   156→   156→    }
   157→   157→  }, [])
   158→   158→
   159→   159→  // ── Геометрия ─────────────────────────────────────────────────────
   160→   160→  const geom = useCallback(() => {
   161→   161→    const w = sizeRef.current.w
   162→   162→    const h = sizeRef.current.h
   163→   163→    const padL = 6, padR = 64, padT = 10, padB = 20
   164→   164→    const plotW = Math.max(30, w - padL - padR)
   165→   165→    const fullH = Math.max(40, h - padT - padB)
   166→   166→    const volH = showVolume ? fullH * 0.16 : 0
   167→   167→    const chartH = fullH - volH - (showVolume ? 8 : 0)
   168→   168→    return { padL, padR, padT, padB, plotW, fullH, volH, chartH }
   169→   169→  }, [showVolume])
   170→   170→
   171→   171→  // ── Главная отрисовка ─────────────────────────────────────────────
   172→   172→  const redraw = useCallback(() => {
   173→   173→    const cv = canvasRef.current
   174→   174→    if (!cv) return
   175→   175→    const ctx = cv.getContext('2d')
   176→   176→    if (!ctx) return
   177→   177→    const dpr = window.devicePixelRatio || 1
   178→   178→    const w = sizeRef.current.w
   179→   179→    const h = sizeRef.current.h
   180→   180→    if (cv.width !== Math.round(w * dpr) || cv.height !== Math.round(h * dpr)) {
   181→   181→      cv.width = Math.round(w * dpr)
   182→   182→      cv.height = Math.round(h * dpr)
   183→   183→    }
   184→   184→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   185→   185→    const C = readColors()
   186→   186→    ctx.clearRect(0, 0, w, h)
   187→   187→    ctx.fillStyle = C.bg
   188→   188→    ctx.fillRect(0, 0, w, h)
   189→   189→
   190→   190→    const cs = candlesRef.current
   191→   191→    if (!cs || cs.length < 2) {
   192→   192→      ctx.fillStyle = C.muted
   193→   193→      ctx.font = '12px ui-monospace, Menlo, monospace'
   194→   194→      ctx.textAlign = 'center'
   195→   195→      ctx.fillText('Загрузка свечей…', w / 2, h / 2)
   196→   196→      return
   197→   197→    }
   198→   198→
   199→   199→    const intervalMs = cs[1].t - cs[0].t || 60_000
   200→   200→    const view = viewRef.current
   201→   201→    view.bars = clamp(view.bars, 15, Math.max(60, cs.length + 40))
   202→   202→    const len = cs.length
   203→   203→    let rightIdx = len - view.offset
   204→   204→    rightIdx = clamp(rightIdx, Math.max(5, view.bars * 0.35), len + view.bars * 0.35)
   205→   205→    const startIdx = rightIdx - view.bars
   206→   206→    const { padL, padR, padT, plotW, volH, chartH, fullH } = geom()
   207→   207→    const barW = plotW / view.bars
   208→   208→    const xOfIdx = (i: number) => padL + (i - startIdx + 0.5) * barW
   209→   209→    const idxOfT = (t: number) => (t - cs[0].t) / intervalMs
   210→   210→    const xOfT = (t: number) => xOfIdx(idxOfT(t))
   211→   211→
   212→   212→    const i0 = Math.max(0, Math.floor(startIdx))
   213→   213→    const i1 = Math.min(len - 1, Math.ceil(rightIdx))
   214→   214→    let minP = Infinity, maxP = -Infinity, maxV = 0
   215→   215→    for (let i = i0; i <= i1; i++) {
   216→   216→      if (cs[i].l < minP) minP = cs[i].l
   217→   217→      if (cs[i].h > maxP) maxP = cs[i].h
   218→   218→      if (cs[i].v > maxV) maxV = cs[i].v
   219→   219→    }
   220→   220→    if (livePrice && livePrice > 0) {
   221→   221→      if (livePrice < minP) minP = livePrice
   222→   222→      if (livePrice > maxP) maxP = livePrice
   223→   223→    }
   224→   224→    const padP = (maxP - minP) * 0.07 || Math.abs(maxP) * 0.01 || 1
   225→   225→    minP -= padP
   226→   226→    maxP += padP
   227→   227→    const yOfP = (p: number) => padT + ((maxP - p) / (maxP - minP)) * chartH
   228→   228→    const pOfY = (y: number) => maxP - ((y - padT) / chartH) * (maxP - minP)
   229→   229→    const tOfX = (x: number) => cs[0].t + ((x - padL) / barW + startIdx - 0.5) * intervalMs
   230→   230→
   231→   231→    // сетка + шкала цены
   232→   232→    ctx.font = '10px ui-monospace, Menlo, monospace'
   233→   233→    ctx.textBaseline = 'middle'
   234→   234→    const step = niceStep(maxP - minP, 6)
   235→   235→    ctx.strokeStyle = C.grid
   236→   236→    ctx.lineWidth = 1
   237→   237→    ctx.fillStyle = C.muted
   238→   238→    ctx.textAlign = 'left'
   239→   239→    for (let p = Math.ceil(minP / step) * step; p <= maxP; p += step) {
   240→   240→      const y = yOfP(p)
   241→   241→      ctx.beginPath()
   242→   242→      ctx.moveTo(padL, Math.round(y) + 0.5)
   243→   243→      ctx.lineTo(padL + plotW, Math.round(y) + 0.5)
   244→   244→      ctx.stroke()
   245→   245→      ctx.fillText(fmtChartPrice(p), padL + plotW + 6, y)
   246→   246→    }
   247→   247→
   248→   248→    // шкала времени
   249→   249→    const labelEvery = Math.max(1, Math.ceil(92 / barW))
   250→   250→    ctx.textAlign = 'center'
   251→   251→    for (let i = i0; i <= i1; i++) {
   252→   252→      if (i % labelEvery !== 0) continue
   253→   253→      const x = xOfIdx(i)
   254→   254→      ctx.strokeStyle = C.grid
   255→   255→      ctx.beginPath()
   256→   256→      ctx.moveTo(Math.round(x) + 0.5, padT)
   257→   257→      ctx.lineTo(Math.round(x) + 0.5, padT + fullH)
   258→   258→      ctx.stroke()
   259→   259→      ctx.fillStyle = C.muted
   260→   260→      ctx.fillText(fmtAxisTime(cs[i].t, interval), x, padT + fullH + 11)
   261→   261→    }
   262→   262→
   263→   263→    // объёмы
   264→   264→    if (showVolume && maxV > 0) {
   265→   265→      const volTop = padT + chartH + 8
   266→   266→      for (let i = i0; i <= i1; i++) {
   267→   267→        const cd = cs[i]
   268→   268→        const x = xOfIdx(i)
   269→   269→        const vh = (cd.v / maxV) * volH
   270→   270→        ctx.fillStyle = cd.c >= cd.o ? `${C.pos}33` : `${C.neg}33`
   271→   271→        ctx.fillRect(x - barW * 0.33, volTop + volH - vh, Math.max(1, barW * 0.66), vh)
   272→   272→      }
   273→   273→    }
   274→   274→
   275→   275→    // свечи
   276→   276→    const bodyW = Math.max(1, barW * 0.66)
   277→   277→    for (let i = i0; i <= i1; i++) {
   278→   278→      const cd = cs[i]
   279→   279→      const x = xOfIdx(i)
   280→   280→      const up = cd.c >= cd.o
   281→   281→      const col = up ? C.pos : C.neg
   282→   282→      ctx.strokeStyle = col
   283→   283→      ctx.lineWidth = 1
   284→   284→      ctx.beginPath()
   285→   285→      ctx.moveTo(Math.round(x) + 0.5, yOfP(cd.h))
   286→   286→      ctx.lineTo(Math.round(x) + 0.5, yOfP(cd.l))
   287→   287→      ctx.stroke()
   288→   288→      const yO = yOfP(cd.o)
   289→   289→      const yC = yOfP(cd.c)
   290→   290→      const top = Math.min(yO, yC)
   291→   291→      const bh = Math.max(1, Math.abs(yC - yO))
   292→   292→      ctx.fillStyle = col
   293→   293→      ctx.fillRect(x - bodyW / 2, top, bodyW, bh)
   294→   294→    }
   295→   295→
   296→   296→    // последняя цена
   297→   297→    const lastC = cs[len - 1]
   298→   298→    const lastP = livePrice && livePrice > 0 ? livePrice : lastC.c
   299→   299→    const lastUp = lastP >= lastC.o
   300→   300→    const yLast = yOfP(lastP)
   301→   301→    ctx.setLineDash([4, 4])
   302→   302→    ctx.strokeStyle = lastUp ? C.pos : C.neg
   303→   303→    ctx.lineWidth = 1
   304→   304→    ctx.beginPath()
   305→   305→    ctx.moveTo(padL, Math.round(yLast) + 0.5)
   306→   306→    ctx.lineTo(padL + plotW, Math.round(yLast) + 0.5)
   307→   307→    ctx.stroke()
   308→   308→    ctx.setLineDash([])
   309→   309→    // чип последней цены
   310→   310→    ctx.fillStyle = lastUp ? C.pos : C.neg
   311→   311→    const lpTxt = fmtChartPrice(lastP)
   312→   312→    ctx.font = 'bold 10px ui-monospace, Menlo, monospace'
   313→   313→    const lpW = ctx.measureText(lpTxt).width + 10
   314→   314→    roundRect(ctx, padL + plotW + 2, yLast - 8, Math.min(lpW, padR - 4), 16, 4)
   315→   315→    ctx.fill()
   316→   316→    ctx.fillStyle = '#0b0e17'
   317→   317→    ctx.textAlign = 'left'
   318→   318→    ctx.fillText(lpTxt, padL + plotW + 7, yLast)
   319→   319→
   320→   320→    // легенда OHLC hovered
   321→   321→    const ms = mouseRef.current
   322→   322→    const hovIdx = ms.inside ? clamp(Math.round(startIdx + (ms.x - padL) / barW - 0.5), 0, len - 1) : len - 1
   323→   323→    const hcd = cs[hovIdx]
   324→   324→    if (hcd) {
   325→   325→      const chg = hcd.o > 0 ? ((hcd.c - hcd.o) / hcd.o) * 100 : 0
   326→   326→      const up = hcd.c >= hcd.o
   327→   327→      ctx.font = '10px ui-monospace, Menlo, monospace'
   328→   328→      ctx.textAlign = 'left'
   329→   329→      ctx.fillStyle = C.muted
   330→   330→      const y0 = padT + 2
   331→   331→      ctx.fillText('O', padL + 4, y0)
   332→   332→      ctx.fillText('H', padL + 62, y0)
   333→   333→      ctx.fillText('L', padL + 130, y0)
   334→   334→      ctx.fillStyle = up ? C.pos : C.neg
   335→   335→      ctx.fillText(fmtChartPrice(hcd.o), padL + 14, y0)
   336→   336→      ctx.fillText(fmtChartPrice(hcd.h), padL + 72, y0)
   337→   337→      ctx.fillText(fmtChartPrice(hcd.l), padL + 140, y0)
   338→   338→      ctx.fillText(`C ${fmtChartPrice(hcd.c)}  ${chg >= 0 ? '+' : ''}${chg.toFixed(2)}%`, padL + 200, y0)
   339→   339→    }
   340→   340→
   341→   341→    // ── Рисунки ────────────────────────────────────────────────────
   342→   342→    const strokeDrawing = (d: ChartDrawing, hover: boolean) => {
   343→   343→      const pts = d.points.map((pt) => ({ x: xOfT(pt.t), y: yOfP(pt.p) }))
   344→   344→      const col = hover ? C.acc : C.acc
   345→   345→      ctx.strokeStyle = col
   346→   346→      ctx.lineWidth = hover ? 2 : 1.5
   347→   347→      ctx.setLineDash([])
   348→   348→      if (d.type === 'trend' && pts.length >= 2) {
   349→   349→        ctx.beginPath()
   350→   350→        ctx.moveTo(pts[0].x, pts[0].y)
   351→   351→        ctx.lineTo(pts[1].x, pts[1].y)
   352→   352→        ctx.stroke()
   353→   353→        dot(pts[0]); dot(pts[1])
   354→   354→      } else if (d.type === 'hline' && pts.length >= 1) {
   355→   355→        const y = pts[0].y
   356→   356→        ctx.beginPath()
   357→   357→        ctx.moveTo(padL, y)
   358→   358→        ctx.lineTo(padL + plotW, y)
   359→   359→        ctx.stroke()
   360→   360→        // чип цены справа
   361→   361→        ctx.fillStyle = col
   362→   362→        ctx.font = 'bold 9px ui-monospace, Menlo, monospace'
   363→   363→        const txt = fmtChartPrice(d.points[0].p)
   364→   364→        const tw = ctx.measureText(txt).width + 8
   365→   365→        roundRect(ctx, padL + plotW + 2, y - 7, Math.min(tw, padR - 4), 14, 3)
   366→   366→        ctx.fill()
   367→   367→        ctx.fillStyle = '#0b0e17'
   368→   368→        ctx.fillText(txt, padL + plotW + 6, y)
   369→   369→      } else if (d.type === 'vline' && pts.length >= 1) {
   370→   370→        const x = pts[0].x
   371→   371→        ctx.setLineDash([2, 3])
   372→   372→        ctx.beginPath()
   373→   373→        ctx.moveTo(x, padT)
   374→   374→        ctx.lineTo(x, padT + fullH)
   375→   375→        ctx.stroke()
   376→   376→        ctx.setLineDash([])
   377→   377→      } else if (d.type === 'rect' && pts.length >= 2) {
   378→   378→        const x = Math.min(pts[0].x, pts[1].x), y = Math.min(pts[0].y, pts[1].y)
   379→   379→        const rw = Math.abs(pts[1].x - pts[0].x), rh = Math.abs(pts[1].y - pts[0].y)
   380→   380→        ctx.fillStyle = `${C.acc}14`
   381→   381→        ctx.fillRect(x, y, rw, rh)

   748→   748→          ms.startX = p.x
   749→   749→          ms.startY = p.y
   750→   750→        }
   751→   751→      } else {
   752→   752→        ms.mode = 'pan'
   753→   753→        ms.startX = p.x
   754→   754→        ms.startY = p.y
   755→   755→        ms.startOffset = view.offset
   756→   756→        cv.style.cursor = 'grabbing'
   757→   757→      }
   758→   758→      redraw()
   759→   759→      return
   760→   760→    }
   761→   761→    if (t === 'ruler') {
   762→   762→      if (ms.mode === 'draw2' && ms.pending) {
   763→   763→        rulerRef.current = { t1: ms.pending.t, p1: ms.pending.p, t2: anchor.t, p2: anchor.p }
   764→   764→        ms.mode = 'idle'
   765→   765→        ms.pending = undefined
   766→   766→        onToolDone?.()
   767→   767→      } else {
   768→   768→        ms.mode = 'draw2'
   769→   769→        ms.pending = anchor
   770→   770→      }
   771→   771→      redraw()
   772→   772→      return
   773→   773→    }
   774→   774→    if (t === 'hline' || t === 'vline') {
   775→   775→      if (chartKey) {
   776→   776→        drawId += 1
   777→   777→        drawingsRef.current = [...drawingsRef.current, { id: `d${Date.now()}_${drawId}`, type: t, points: [anchor] }]
   778→   778→        saveDrawings(chartKey, drawingsRef.current)
   779→   779→      }
   780→   780→      onToolDone?.()
   781→   781→      redraw()
   782→   782→      return
   783→   783→    }
   784→   784→    // двухточечные: тренд / прямоугольник / фибо
   785→   785→    if (ms.mode === 'draw2' && ms.pending) {
   786→   786→      if (chartKey) {
   787→   787→        drawId += 1
   788→   788→        drawingsRef.current = [...drawingsRef.current, { id: `d${Date.now()}_${drawId}`, type: t as DrawType, points: [ms.pending, anchor] }]
   789→   789→        saveDrawings(chartKey, drawingsRef.current)
   790→   790→      }
   791→   791→      ms.mode = 'idle'
   792→   792→      ms.pending = undefined
   793→   793→      onToolDone?.()
   794→   794→    } else {
   795→   795→      ms.mode = 'draw2'
   796→   796→      ms.pending = anchor
   797→   797→    }
   798→   798→    redraw()
   799→   799→  }
   800→   800→
   801→   801→  const onMouseUp = () => {
   802→   802→    const ms = mouseRef.current
   803→   803→    if (ms.mode === 'ruler' && ms.pending) {
   804→   804→      // линейка колёсиком: фиксируем и оставляем до Esc/следующего замера
   805→   805→      rulerRef.current = { t1: ms.pending.t, p1: ms.pending.p, t2: ms.x ? tOfXFromState(ms.x) : ms.pending.t, p2: pOfYFromState(ms.y) }
   806→   806→      ms.pending = undefined
   807→   807→      ms.mode = 'idle'
   808→   808→    } else if (ms.mode === 'pan' || ms.mode === 'drag') {
   809→   809→      ms.mode = 'idle'
   810→   810→      ms.grab = undefined
   811→   811→      const cv = canvasRef.current
   812→   812→      if (cv) cv.style.cursor = 'crosshair'
   813→   813→    }
   814→   814→    redraw()
   815→   815→  }

   817→   817→  // хелперы цены/времени под курсором (для фиксирования линейки)
   818→   818→  function pOfYFromState(y: number): number {
   819→   819→    const cs = candlesRef.current
   820→   820→    if (!cs || cs.length < 2) return 0
   821→   821→    const view = viewRef.current
   822→   822→    const len = cs.length
   823→   823→    const rightIdx = clamp(len - view.offset, Math.max(5, view.bars * 0.35), len + view.bars * 0.35)
   824→   824→    const startIdx = rightIdx - view.bars
   825→   825→    const { chartH } = geom()
   826→   826→    let minP = Infinity, maxP = -Infinity
   827→   827→    const i0 = Math.max(0, Math.floor(startIdx)), i1 = Math.min(len - 1, Math.ceil(rightIdx))
   828→   828→    for (let i = i0; i <= i1; i++) { if (cs[i].l < minP) minP = cs[i].l; if (cs[i].h > maxP) maxP = cs[i].h }
   829→   829→    const padP = (maxP - minP) * 0.07 || 1
   830→   830→    minP -= padP; maxP += padP
   831→   831→    return maxP - ((y - 10) / chartH) * (maxP - minP)
   832→   832→  }
   833→   833→  function tOfXFromState(x: number): number {
   834→   834→    const cs = candlesRef.current
   835→   835→    if (!cs || cs.length < 2) return Date.now()
   836→   836→    const intervalMs = cs[1].t - cs[0].t || 60_000
   837→   837→    const view = viewRef.current
   838→   838→    const len = cs.length
   839→   839→    const rightIdx = clamp(len - view.offset, Math.max(5, view.bars * 0.35), len + view.bars * 0.35)
   840→   840→    const startIdx = rightIdx - view.bars
   841→   841→    const { padL, plotW } = geom()
   842→   842→    const barW = plotW / view.bars
   843→   843→    return cs[0].t + ((x - padL) / barW + startIdx - 0.5) * intervalMs
   844→   844→  }

   846→   846→  const onWheelNative = useCallback((e: WheelEvent) => {
   847→   847→    if (!interactive) return
   848→   848→    e.preventDefault()
   849→   849→    const cs = candlesRef.current
   850→   850→    if (!cs || cs.length < 2) return
   851→   851→    const view = viewRef.current
   852→   852→    const { padL, plotW } = geom()
   853→   853→    const cv = canvasRef.current
   854→   854→    if (!cv) return
   855→   855→    const r = cv.getBoundingClientRect()
   856→   856→    const px = e.clientX - r.left
   857→   857→    const factor = e.deltaY > 0 ? 1.15 : 1 / 1.15
   858→   858→    const oldBars = view.bars
   859→   859→    const newBars = clamp(oldBars * factor, 15, Math.max(60, cs.length + 40))
   860→   860→    // якорь: индекс под курсором остаётся на месте
   861→   861→    const len = cs.length
   862→   862→    const oldRight = clamp(len - view.offset, Math.max(5, oldBars * 0.35), len + oldBars * 0.35)
   863→   863→    const oldStart = oldRight - oldBars
   864→   864→    const anchorIdx = oldStart + ((px - padL) / plotW) * oldBars
   865→   865→    const newRight = anchorIdx + (1 - (px - padL) / plotW) * newBars
   866→   866→    view.bars = newBars
   867→   867→    view.offset = len - newRight
   868→   868→    redrawRef.current()
   869→   869→  }, [geom, interactive])
   870→   870→
   871→   871→  // нативный non-passive wheel: зум колесом без прокрутки страницы
   872→   872→  useEffect(() => {
   873→   873→    const cv = canvasRef.current
   874→   874→    if (!cv) return
   875→   875→    cv.addEventListener('wheel', onWheelNative, { passive: false })
   876→   876→    return () => cv.removeEventListener('wheel', onWheelNative)
   877→   877→  }, [onWheelNative])
   878→   878→
   879→   879→  const onDoubleClick = () => {
   880→   880→    viewRef.current = { bars: 130, offset: 0 }
   881→   881→    redraw()
   882→   882→  }
   883→   883→
   884→   884→  const onLeave = () => {
   885→   885→    const ms = mouseRef.current
   886→   886→    ms.inside = false
   887→   887→    // драг/линейка завершились вне канваса — фиксируем
   888→   888→    if (ms.mode === 'ruler' && ms.pending) {
   889→   889→      rulerRef.current = { t1: ms.pending.t, p1: ms.pending.p, t2: tOfXFromState(ms.x), p2: pOfYFromState(ms.y) }
   890→   890→      ms.pending = undefined
   891→   891→      ms.mode = 'idle'
   892→   892→    } else if (ms.mode === 'pan' || ms.mode === 'drag') {
   893→   893→      ms.mode = 'idle'
   894→   894→      ms.grab = undefined
   895→   895→      const cv = canvasRef.current
   896→   896→      if (cv) cv.style.cursor = 'crosshair'
   897→   897→    }
   898→   898→    redraw()
   899→   899→  }
   900→   900→
   901→   901→  return (
   902→   902→    <div ref={wrapRef} className={cn('relative h-full w-full', className)}>
   903→   903→      <canvas
   904→   904→        ref={canvasRef}
   905→   905→        className="absolute inset-0 h-full w-full"
   906→   906→        style={{ touchAction: 'none' }}
   907→   907→        onMouseMove={onMouseMove}
   908→   908→        onMouseDown={onMouseDown}
   909→   909→        onMouseUp={onMouseUp}
   910→   910→        onMouseLeave={onLeave}
   911→   911→        onDoubleClick={onDoubleClick}
   912→   912→        onContextMenu={(e) => e.preventDefault()}
   913→   913→      />
   914→   914→    </div>
   915→   915→  )
   916→   916→}
