    38→// ─── Загрузка символов биржи ─────────────────────────────────────────
    39→
    40→function useSymbols(ex: ExchangeId, market: MarketType | 'ALL') {
    41→  const [rows, setRows] = useState<SymbolRow[]>([])
    42→  const [loading, setLoading] = useState(true)
    43→  useEffect(() => {
    44→    let alive = true
    45→    const hub = getHub()
    46→    const load = async () => {
    47→      // первый await выводит из синхронного контекста эффекта
    48→      await Promise.resolve()
    49→      if (!alive) return
    50→      setLoading(true)
    51→      if (market === 'ALL') {
    52→        const [f, s] = await Promise.all([hub.fetchSymbols(ex, 'F'), hub.fetchSymbols(ex, 'S')])
    53→        if (!alive) return
    54→        const map = new Map<string, SymbolRow>()
    55→        for (const r of [...f, ...s]) {
    56→          const prev = map.get(r.symbol)
    57→          if (!prev || r.vol24h > prev.vol24h) map.set(r.symbol, r)
    58→        }
    59→        setRows([...map.values()].sort((a, b) => b.vol24h - a.vol24h))
    60→      } else {
    61→        setRows(await hub.fetchSymbols(ex, market))
    62→      }
    63→      if (alive) setLoading(false)
    64→    }
    65→    void load()
    66→    return () => { alive = false }
    67→  }, [ex, market])
    68→  return { rows, loading }
    69→}
