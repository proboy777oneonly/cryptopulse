    35→function NumberField({ label, value, onChange, unit, step = 1 }: {

   193→    () => [...hub.sim.coinsOn(win.exchange, 'F'), ...hub.sim.coinsOn(win.exchange, 'S')].filter((s, i, a) => a.indexOf(s) === i).sort(),

   207→          </DialogTitle>

   226→        </div>

   335→function BlacklistDialog({ win, open, onOpenChange }: { win: DensityWindowConfig; open: boolean; onOpenChange: (v: boolean) => void }) {

   342→  const list = tab === 'F' ? win.blacklistF : win.blacklistS
   343→
   344→  return (

   351→          </DialogTitle>
   352→        </DialogHeader>
   353→        <Tabs value={tab} onValueChange={(v) => setTab(v as 'F' | 'S')}>
   354→          <TabsList className="grid w-full grid-cols-2">
   355→            <TabsTrigger value="F">Фьючерсы ({win.blacklistF.length})</TabsTrigger>
   356→            <TabsTrigger value="S">Спот ({win.blacklistS.length})</TabsTrigger>

   358→        </Tabs>

   361→            <SelectTrigger className="flex-1"><SelectValue placeholder="Монета для блокировки" /></SelectTrigger>
   362→            <SelectContent className="max-h-56">
   363→              {available.filter((s) => !list.includes(s)).map((s) => (
   364→                <SelectItem key={s} value={s}>{s} · {COIN_MAP[s]?.name ?? ''}</SelectItem>
   365→              ))}
   366→            </SelectContent>
   367→          </Select>

   369→            <BanIcon className="mr-1 h-3.5 w-3.5" /> В список
   370→          </Button>

   374→          {list.map((s) => (
   375→            <span key={s} className="flex items-center gap-1.5 rounded-lg border px-2.5 py-1 text-xs font-bold" style={{ borderColor: 'color-mix(in srgb, var(--destructive) 40%, transparent)', background: 'color-mix(in srgb, var(--destructive) 8%, transparent)', color: 'var(--destructive)' }}>

   521→function DensityRow({ e, compact, onSend }: { e: DensityEvent; compact: boolean; onSend: () => void }) {
   522→  const now = Date.now()
   523→  const left = Math.max(0, (e.expiresAt - now) / (e.lifetimeSec * 1000))
   524→  const color = EX_MAP[e.exchange].color
   525→  return (
   526→    <motion.div
   527→      layout="position"
   528→      initial={{ opacity: 0, y: -8 }}
   529→      animate={{ opacity: 1, y: 0 }}
   530→      exit={{ opacity: 0 }}
   531→      className={cn('group flex items-center gap-2 rounded-lg border border-[var(--border)] bg-[var(--muted)] px-2.5 transition-colors hover:border-[color-mix(in_srgb,var(--acc)_35%,transparent)]', compact ? 'py-1' : 'py-1.5')}
   532→    >
   533→      <span className="h-1.5 w-1.5 shrink-0 rounded-full" style={{ background: color }} />
   534→      <span className="w-12 shrink-0 truncate text-xs font-black">{e.symbol}</span>
   535→      <span
   536→        className="shrink-0 rounded px-1 py-0.5 text-[8px] font-black"
   537→        style={{ background: e.market === 'F' ? '#f59e0b1e' : '#22d3ee1e', color: e.market === 'F' ? '#f59e0b' : '#22d3ee' }}
   538→      >
   539→        {e.market}
   540→      </span>
   541→      <span className="cp-num w-14 shrink-0 text-xs font-semibold">{fmtUsd(e.volumeUsd)}</span>
   542→      <span className="cp-num w-11 shrink-0 text-right text-xs font-bold" style={{ color: 'var(--acc)' }}>{e.strengthPct.toFixed(2)}%</span>
   543→      <span className="w-12 shrink-0 text-right text-[10px] font-black" style={{ color: e.direction === 'LONG' ? 'var(--pos)' : 'var(--neg)' }}>
   544→        {e.direction}
   545→      </span>
   546→      {!compact && <span className="cp-num hidden w-11 shrink-0 text-right text-[10px] text-[var(--muted-foreground)] md:block">{fmtTime(e.createdAt)}</span>}
   547→      <div className="hidden h-1 w-10 shrink-0 overflow-hidden rounded-full bg-[var(--border)] md:block" title={`Осталось ${Math.ceil(left * e.lifetimeSec)}с`}>
   548→        <div className="h-full rounded-full transition-all duration-1000" style={{ width: `${left * 100}%`, background: left > 0.4 ? 'var(--pos)' : 'var(--neg)' }} />
   549→      </div>
   550→      <button
   551→        onClick={onSend}
   552→        className="ml-auto flex h-6 w-6 shrink-0 items-center justify-center rounded-md border border-transparent text-[var(--muted-foreground)] opacity-0 transition-all group-hover:opacity-100 hover:border-[color-mix(in_srgb,var(--acc)_50%,transparent)] hover:text-[var(--acc)]"
   553→        title="Отправить тикер в другие окна"
   554→        aria-label="Отправить тикер"
   555→      >
   556→        <Send className="h-3 w-3" />
   557→      </button>
   558→    </motion.div>
   559→  )
   560→}
